// web/src/core/api/httpClient.ts

export type HttpMethod = "GET" | "POST" | "PUT" | "PATCH" | "DELETE";

export type HttpRequestConfig = {
  method?: HttpMethod;
  headers?: Record<string, string>;
  body?: unknown;
  signal?: AbortSignal;
  credentials?: RequestCredentials;
  skipAuthRetry?: boolean;
};

export type ApiErrorCode =
  | "NAO_AUTORIZADO"
  | "NAO_AUTENTICADO"
  | "INVALIDO"
  | "NAO_ENCONTRADO"
  | "ERRO_INTERNO"
  | "CONFLITO";

export type ApiErrorEnvelope = {
  erro?: {
    codigo?: string;
    mensagem?: string;
    detalhes?: Record<string, unknown>;
  };
};

export class HttpError extends Error {
  public readonly status: number;
  public readonly code?: string;
  public readonly details?: Record<string, unknown>;

  constructor(params: {
    message: string;
    status: number;
    code?: string;
    details?: Record<string, unknown>;
  }) {
    super(params.message);
    this.name = "HttpError";
    this.status = params.status;
    this.code = params.code;
    this.details = params.details;
  }
}

const API_BASE_URL = "http://localhost:3333/api";

let accessTokenGetter: (() => string | null) | null = null;
let refreshSessionHandler: (() => Promise<boolean>) | null = null;
let unauthorizedHandler: (() => void) | null = null;

export function registerAccessTokenGetter(getter: () => string | null): void {
  accessTokenGetter = getter;
}

export function registerRefreshSessionHandler(handler: (() => Promise<boolean>) | null): void {
  refreshSessionHandler = handler;
}

export function registerUnauthorizedHandler(handler: (() => void) | null): void {
  unauthorizedHandler = handler;
}

function buildUrl(path: string): string {
  if (/^https?:\/\//i.test(path)) {
    return path;
  }

  return `${API_BASE_URL}${path.startsWith("/") ? path : `/${path}`}`;
}

function isJsonResponse(response: Response): boolean {
  const contentType = response.headers.get("content-type") ?? "";
  return contentType.includes("application/json");
}

async function buildHttpError(response: Response): Promise<HttpError> {
  if (isJsonResponse(response)) {
    const data = (await response.json()) as ApiErrorEnvelope;

    return new HttpError({
      message: data.erro?.mensagem?.trim() || `Erro HTTP ${response.status}`,
      status: response.status,
      code: data.erro?.codigo,
      details: data.erro?.detalhes,
    });
  }

  const text = await response.text().catch(() => "");
  return new HttpError({
    message: text || `Erro HTTP ${response.status}`,
    status: response.status,
  });
}

function isFormDataBody(body: unknown): body is FormData {
  return typeof FormData !== "undefined" && body instanceof FormData;
}

function buildHeaders(config: HttpRequestConfig, token: string | null): Record<string, string> {
  const headers: Record<string, string> = {
    Accept: "application/json",
    ...config.headers,
  };

  const hasBody = config.body !== undefined && config.body !== null;

  if (hasBody && !isFormDataBody(config.body)) {
    headers["Content-Type"] = "application/json";
  }

  // Só insere o header Authorization quando há token e não houver
  // um header `Authorization` explícito (ou quando ele for uma string vazia
  // usada para desabilitar a injeção automática, por exemplo em refresh/logout).
  // if (token && (headers.Authorization === undefined || headers.Authorization === "")) {
if (token && headers.Authorization !== "") {
  headers.Authorization = `Bearer ${token}`;
}

  return headers;
}

function buildBody(body: unknown): BodyInit | undefined {
  if (body === undefined || body === null) {
    return undefined;
  }

  if (isFormDataBody(body)) {
    return body;
  }

  return JSON.stringify(body);
}

async function parseSuccessResponse<TResponse>(response: Response): Promise<TResponse> {
  if (response.status === 204) {
    return undefined as TResponse;
  }

  if (isJsonResponse(response)) {
    return (await response.json()) as TResponse;
  }

  return (await response.text()) as TResponse;
}

let refreshPromise: Promise<boolean> | null = null;

async function ensureRefreshedSession(): Promise<boolean> {
  if (!refreshSessionHandler) {
    return false;
  }

  if (!refreshPromise) {
    refreshPromise = (async () => {
      try {
        return await refreshSessionHandler();
      } finally {
        refreshPromise = null;
      }
    })();
  }

  return refreshPromise;
}

async function rawFetch(path: string, config: HttpRequestConfig): Promise<Response> {
  const token = accessTokenGetter?.() ?? null;

  return fetch(buildUrl(path), {
    method: config.method ?? "GET",
    headers: buildHeaders(config, token),
    body: buildBody(config.body),
    signal: config.signal,
    credentials: config.credentials ?? "include",
  });
}

export async function httpRequest<TResponse>(path: string, config: HttpRequestConfig = {}): Promise<TResponse> {
  let response = await rawFetch(path, config);

  if (response.status === 401 && !config.skipAuthRetry) {
    const refreshed = await ensureRefreshedSession();

    if (refreshed) {
      response = await rawFetch(path, {
        ...config,
        skipAuthRetry: true,
      });
    } else {
      unauthorizedHandler?.();
    }
  }

  if (!response.ok) {
    throw await buildHttpError(response);
  }

  return parseSuccessResponse<TResponse>(response);
}

export const httpClient = {
  get<TResponse>(path: string, config?: Omit<HttpRequestConfig, "method" | "body">) {
    return httpRequest<TResponse>(path, {
      ...config,
      method: "GET",
    });
  },

  post<TResponse, TBody = unknown>(path: string, body?: TBody, config?: Omit<HttpRequestConfig, "method" | "body">) {
    return httpRequest<TResponse>(path, {
      ...config,
      method: "POST",
      body,
    });
  },

  put<TResponse, TBody = unknown>(path: string, body?: TBody, config?: Omit<HttpRequestConfig, "method" | "body">) {
    return httpRequest<TResponse>(path, {
      ...config,
      method: "PUT",
      body,
    });
  },

  patch<TResponse, TBody = unknown>(path: string, body?: TBody, config?: Omit<HttpRequestConfig, "method" | "body">) {
    return httpRequest<TResponse>(path, {
      ...config,
      method: "PATCH",
      body,
    });
  },

  delete<TResponse>(path: string, config?: Omit<HttpRequestConfig, "method" | "body">) {
    return httpRequest<TResponse>(path, {
      ...config,
      method: "DELETE",
    });
  },
};

// web/src/core/api/permissions.ts

import type { StatusUsuario } from "../../features/tenant-admin/usuarios/tipos";

export const USUARIOS_PERMISSIONS = {
  LISTAR: "tenant.usuario.listar",
  CRIAR: "tenant.usuario.criar",
  ATUALIZAR: "tenant.usuario.atualizar",
} as const;

export const USUARIOS_MODULE = "USUARIOS" as const;

export type UsuarioAction =
  | "listar"
  | "ver"
  | "criar"
  | "editar"
  | "inativar"
  | "reativar";

export type FrontendAuthorizationContext = {
  permissions?: string[];
  modules?: string[];
  isSuperAdmin?: boolean;
};

function normalizarLista(values?: string[]) {
  return (values ?? []).map((value) => value.trim().toUpperCase());
}

export function hasPermission(
  ctx: FrontendAuthorizationContext,
  permission: string,
): boolean {
  if (ctx.isSuperAdmin) return true;
  return normalizarLista(ctx.permissions).includes(permission.trim().toUpperCase());
}

export function hasModule(
  ctx: FrontendAuthorizationContext,
  moduleCode: string,
): boolean {
  if (ctx.isSuperAdmin) return true;
  return normalizarLista(ctx.modules).includes(moduleCode.trim().toUpperCase());
}

export function canAccessUsuariosModule(
  ctx: FrontendAuthorizationContext,
): boolean {
  return (
    hasModule(ctx, USUARIOS_MODULE) &&
    hasPermission(ctx, USUARIOS_PERMISSIONS.LISTAR)
  );
}

export function canCreateUsuario(
  ctx: FrontendAuthorizationContext,
): boolean {
  return (
    hasModule(ctx, USUARIOS_MODULE) &&
    hasPermission(ctx, USUARIOS_PERMISSIONS.CRIAR)
  );
}

export function canEditUsuario(
  ctx: FrontendAuthorizationContext,
): boolean {
  return (
    hasModule(ctx, USUARIOS_MODULE) &&
    hasPermission(ctx, USUARIOS_PERMISSIONS.ATUALIZAR)
  );
}

export function canToggleUsuarioStatus(
  ctx: FrontendAuthorizationContext,
  statusAtual: StatusUsuario,
): boolean {
  if (
    !hasModule(ctx, USUARIOS_MODULE) ||
    !hasPermission(ctx, USUARIOS_PERMISSIONS.ATUALIZAR)
  ) {
    return false;
  }

  return statusAtual === "ATIVO" || statusAtual === "INATIVO";
}

export function canRunUsuarioAction(
  ctx: FrontendAuthorizationContext,
  action: UsuarioAction,
  statusAtual?: StatusUsuario,
): boolean {
  switch (action) {
    case "listar":
    case "ver":
      return canAccessUsuariosModule(ctx);
    case "criar":
      return canCreateUsuario(ctx);
    case "editar":
      return canEditUsuario(ctx);
    case "inativar":
    case "reativar":
      return statusAtual ? canToggleUsuarioStatus(ctx, statusAtual) : false;
    default:
      return false;
  }
}

import { describe, it, expect, vi, beforeEach } from "vitest";

function TestComponent(): JSX.Element {
  const { status, isAuthenticated, user } = useAuth();

  return (
    <div>
      <span data-testid="status">{status}</span>
      <span data-testid="is-authenticated">
        {isAuthenticated ? "true" : "false"}
      </span>
      <span data-testid="user-id">{user?.usuarioId ?? ""}</span>

  return <AuthProvider>{children}</AuthProvider>;

describe("AuthContext", () => {
  beforeEach(() => {
    vi.clearAllMocks();
  });

  it("deve iniciar como não autenticado quando não houver token", async () => {
    vi.mocked(authStorage.getAccessToken).mockReturnValue(null);

    render(<TestComponent />, { wrapper: Wrapper });

    await waitFor(() => {
      expect(screen.getByTestId("status").textContent).toBe("unauthenticated");
      expect(screen.getByTestId("is-authenticated").textContent).toBe("false");
    });

      expect(authStorage.clear).toHaveBeenCalled();
      expect(screen.getByTestId("status").textContent).toBe("unauthenticated");
    });

// apps/web/src/core/auth/authContext.tsx

import { createContext, useCallback, useEffect, useMemo, useState, type PropsWithChildren, type JSX } from "react";
import { registerAccessTokenGetter, registerRefreshSessionHandler, registerUnauthorizedHandler } from "../api/httpClient";
import { authStorage } from "./authStorage";
import { login as loginRequest, loginPlataforma as loginPlataformaRequest, logout as logoutRequest, me as meRequest, 
  refresh as refreshRequest } from "../../features/auth/api/authApi";
import type { AuthContextValue, LoginInput, LoginPlataformaInput, SessionUser } from "./tipos";

export const AuthContext = createContext<AuthContextValue | undefined>(undefined);

function normalizeUser(user: SessionUser): SessionUser {
  const perfil =
    user.perfil ??
    (user.locatarioId ? "TENANT" : "PLATAFORMA");

  return {
    ...user,
    usuarioId: user.usuarioId,
    locatarioId: user.locatarioId ?? null,
    perfil,
  };
}

export function AuthProvider({ children }: PropsWithChildren): JSX.Element {
  const [status, setStatus] = useState<AuthContextValue["status"]>("idle");
  const [accessToken, setAccessToken] = useState<string | null>(
    authStorage.getAccessToken(),
  );
  const [user, setUser] = useState<SessionUser | null>(authStorage.getUser());

  const clearSession = useCallback(() => {
    authStorage.clear();
    setAccessToken(null);
    setUser(null);
    setStatus("unauthenticated");
  }, []);

  const applySession = useCallback((params: {
    accessToken: string;
    refreshToken?: string;
    usuario: SessionUser;
  }): SessionUser => {
    const normalizedUser = normalizeUser(params.usuario);

    authStorage.setAccessToken(params.accessToken);
    authStorage.setUser(normalizedUser);
    authStorage.setLocatarioId(normalizedUser.locatarioId);

    if (params.refreshToken) {
      authStorage.setRefreshToken(params.refreshToken);
    } else {
      authStorage.removeRefreshToken();
    }

    setAccessToken(params.accessToken);
    setUser(normalizedUser);
    setStatus("authenticated");

    return normalizedUser;
  }, []);

  const carregarUsuarioAutenticado = useCallback(async (): Promise<SessionUser> => {
    const meResponse = await meRequest();
    const normalizedUser = normalizeUser(meResponse.usuario);

    authStorage.setUser(normalizedUser);
    authStorage.setLocatarioId(normalizedUser.locatarioId);
    setUser(normalizedUser);

    return normalizedUser;
  }, []);

  const refreshTokens = useCallback(async (): Promise<boolean> => {
    const currentUser = authStorage.getUser();

    if (currentUser?.perfil === "PLATAFORMA") {
      return false;
    }

    const refreshToken = authStorage.getRefreshToken();
    const locatarioId =
      authStorage.getLocatarioId() ?? currentUser?.locatarioId ?? null;

    if (!refreshToken) {
      return false;
    }

    try {
      const response = await refreshRequest({
        refreshToken,
        locatarioId,
      });

      authStorage.setAccessToken(response.accessToken);
      setAccessToken(response.accessToken);

      if (response.refreshToken) {
        authStorage.setRefreshToken(response.refreshToken);
      }

      return true;
    } catch {
      clearSession();
      return false;
    }
  }, [clearSession]);

  const login = useCallback(
    async (input: LoginInput): Promise<SessionUser> => {
      setStatus("loading");

      try {
        const response = await loginRequest(input);

        // O login tenant retorna tokens. O usuário da sessão vem de /auth/me.
        // Por isso, persistimos temporariamente o access token antes de chamar meRequest().
        authStorage.setAccessToken(response.accessToken);
        setAccessToken(response.accessToken);

        if (response.refreshToken) {
          authStorage.setRefreshToken(response.refreshToken);
        } else {
          authStorage.removeRefreshToken();
        }

        const usuario =
          (response as { usuario?: SessionUser }).usuario ??
          (await meRequest()).usuario;

        return applySession({
          accessToken: response.accessToken,
          refreshToken: response.refreshToken,
          usuario,
        });
      } catch (error) {
        clearSession();
        throw error;
      }
    },
    [applySession, clearSession],
  );

  const loginPlataforma = useCallback(
    async (input: LoginPlataformaInput): Promise<SessionUser> => {
      setStatus("loading");

      try {
        const response = await loginPlataformaRequest(input);

        return applySession({
          accessToken: response.accessToken,
          usuario: response.usuario,
        });
      } catch (error) {
        clearSession();
        throw error;
      }
    },
    [applySession, clearSession],
  );

  const logout = useCallback(async (): Promise<void> => {
    const refreshToken = authStorage.getRefreshToken();
    const locatarioId = authStorage.getLocatarioId();

    try {
      if (refreshToken && user?.perfil === "TENANT") {
        await logoutRequest({
          refreshToken,
          locatarioId,
        });
      }
    } finally {
      clearSession();
    }
  }, [clearSession, user]);

  useEffect(() => {
    registerAccessTokenGetter(() => authStorage.getAccessToken());
    registerRefreshSessionHandler(refreshTokens);
    registerUnauthorizedHandler(clearSession);

    return () => {
      registerAccessTokenGetter(() => null);
      registerRefreshSessionHandler(null);
      registerUnauthorizedHandler(null);
    };
  }, [clearSession, refreshTokens]);

  useEffect(() => {
    let ativo = true;

    async function bootstrap() {
      const token = authStorage.getAccessToken();

      if (!token) {
        if (ativo) {
          setStatus("unauthenticated");
        }
        return;
      }

      setStatus("loading");

      try {
        await carregarUsuarioAutenticado();

        if (ativo) {
          setStatus("authenticated");
        }
      } catch {
        if (ativo) {
          clearSession();
        }
      }
    }

    void bootstrap();

    return () => {
      ativo = false;
    };
  }, [carregarUsuarioAutenticado, clearSession]);

  const value = useMemo<AuthContextValue>(
    () => ({
      status,
      user,
      accessToken,
      isLoading: status === "idle" || status === "loading",
      isAuthenticated: status === "authenticated" && Boolean(user),
      login,
      loginPlataforma,
      logout,
      refreshTokens,
    }),
    [status, user, accessToken, login, loginPlataforma, logout, refreshTokens],
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

// apps/web/src/core/auth/authStorage.ts

import type { SessionUser } from "./tipos";

const ACCESS_TOKEN_KEY = "app.access_token";
const REFRESH_TOKEN_KEY = "app.refresh_token";
const LOCATARIO_ID_KEY = "app.locatario_id";
const USER_KEY = "app.auth_user";

function safeGet(key: string): string | null {
  try {
    return localStorage.getItem(key);
  } catch {
    return null;
  }
}

function safeSet(key: string, value: string): void {
  try {
    localStorage.setItem(key, value);
  } catch {
    // storage não deve quebrar a aplicação
  }
}

function safeRemove(key: string): void {
  try {
    localStorage.removeItem(key);
  } catch {
    // noop
  }
}

export const authStorage = {
  getAccessToken(): string | null {
    return safeGet(ACCESS_TOKEN_KEY);
  },

  setAccessToken(token: string): void {
    safeSet(ACCESS_TOKEN_KEY, token);
  },

  removeAccessToken(): void {
    safeRemove(ACCESS_TOKEN_KEY);
  },

  getRefreshToken(): string | null {
    return safeGet(REFRESH_TOKEN_KEY);
  },

  setRefreshToken(token: string): void {
    safeSet(REFRESH_TOKEN_KEY, token);
  },

  removeRefreshToken(): void {
    safeRemove(REFRESH_TOKEN_KEY);
  },

  getLocatarioId(): string | null {
    return safeGet(LOCATARIO_ID_KEY);
  },

  setLocatarioId(locatarioId: string | null): void {
    if (locatarioId) {
      safeSet(LOCATARIO_ID_KEY, locatarioId);
    } else {
      safeRemove(LOCATARIO_ID_KEY);
    }
  },

  getUser(): SessionUser | null {
    const raw = safeGet(USER_KEY);

    if (!raw) return null;

    try {
      return JSON.parse(raw) as SessionUser;
    } catch {
      safeRemove(USER_KEY);
      return null;
    }
  },

  setUser(user: SessionUser): void {
    safeSet(USER_KEY, JSON.stringify(user));
  },

  removeUser(): void {
    safeRemove(USER_KEY);
  },

  clear(): void {
    safeRemove(ACCESS_TOKEN_KEY);
    safeRemove(REFRESH_TOKEN_KEY);
    safeRemove(LOCATARIO_ID_KEY);
    safeRemove(USER_KEY);
  },
};

===== apps/web/src/core/errors/mapError.ts =====

// frontend/src/core/errors/mapError.ts

import { HttpError } from "../api/httpClient";
import type { AppErrorCode } from "./errorCodes";

export type UiError = {
  code: AppErrorCode | "UNKNOWN";
  title: string;
  message: string;
  status?: number;
};

function buildDefaultMessageByCode(code: AppErrorCode): UiError {
  switch (code) {
    case "NAO_AUTENTICADO":
      return {
        code,
        title: "Sessão inválida",
        message: "Sua sessão expirou ou é inválida. Faça login novamente.",
      };

    case "NAO_AUTORIZADO":
      return {
        code,
        title: "Acesso negado",
        message: "Você não possui permissão para realizar esta ação.",
      };

    case "INVALIDO":
      return {
        code,
        title: "Dados inválidos",
        message: "Verifique os dados informados e tente novamente.",
      };

    case "NAO_ENCONTRADO":
      return {
        code,
        title: "Não encontrado",
        message: "O recurso solicitado não foi encontrado.",
      };

    case "CONFLITO":
      return {
        code,
        title: "Conflito",
        message: "Não foi possível concluir a operação por conflito de dados.",
      };

    case "ERRO_INTERNO":
      return {
        code,
        title: "Erro interno",
        message: "Ocorreu um erro interno. Tente novamente mais tarde.",
      };
    default:
      return {
        code: "UNKNOWN",
        title: "Erro",
        message: "Ocorreu um erro inesperado.",
      };
  }
}

export function mapError(error: unknown): UiError {
  if (error instanceof HttpError) {
    const code = error.code as AppErrorCode | undefined;

    if (code) {
      const mapped = buildDefaultMessageByCode(code);

      return {
        ...mapped,
        status: error.status,
        message: error.message?.trim() || mapped.message,
      };
    }

    if (error.status === 401) {
      return {
        code: "NAO_AUTENTICADO",
        title: "Sessão inválida",
        message: "Sua sessão expirou ou é inválida. Faça login novamente.",
        status: error.status,
      };
    }

    if (error.status === 403) {
      return {
        code: "NAO_AUTORIZADO",
        title: "Acesso negado",
        message: "Você não possui permissão para acessar este recurso.",
        status: error.status,
      };
    }

    if (error.status === 404) {
      return {
        code: "NAO_ENCONTRADO",
        title: "Não encontrado",
        message: "O recurso solicitado não foi encontrado.",
        status: error.status,
      };
    }

    if (error.status === 409) {
      return {
        code: "CONFLITO",
        title: "Conflito",
        message: "Não foi possível concluir a operação por conflito de dados.",
        status: error.status,
      };
    }

    if (error.status >= 400 && error.status < 500) {
      return {
        code: "INVALIDO",
        title: "Solicitação inválida",
        message: error.message || "A solicitação não pôde ser processada.",
        status: error.status,
      };
    }

    return {
      code: "ERRO_INTERNO",
      title: "Erro interno",
      message: error.message || "Ocorreu um erro interno.",
      status: error.status,
    };
  }

  if (error instanceof Error) {
    return {
      code: "UNKNOWN",
      title: "Erro inesperado",
      message: error.message || "Ocorreu um erro inesperado.",
    };
  }

  return {
    code: "UNKNOWN",
    title: "Erro inesperado",
    message: "Ocorreu um erro inesperado.",
  };
}

// apps/web/src/core/routing/pageGuard.tsx

import type { PropsWithChildren, ReactNode } from "react";
import { useAuthorization } from "../auth/useAuthorization";

type PageGuardProps = PropsWithChildren<{
  requiredPermissions?: string[];
  requiredModule?: string;
  requireAllPermissions?: boolean;
  fallback?: ReactNode;
}>;

function hasRequiredPermissions(currentPermissions: string[], requiredPermissions: string[], requireAllPermissions: boolean): boolean {
  if (requiredPermissions.length === 0) {
    return true;
  }

  if (requireAllPermissions) {
    return requiredPermissions.every((permission) =>
      currentPermissions.includes(permission),
    );
  }

  return requiredPermissions.some((permission) =>
    currentPermissions.includes(permission),
  );
}

export function PageGuard({ children, requiredPermissions = [], requiredModule, requireAllPermissions = true, fallback }: PageGuardProps) {
  const { hasModule, permissions } = useAuthorization();

  if (requiredModule && !hasModule(requiredModule)) {
    return (
      fallback ?? (
        <div className="rounded-xl border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-800">
          Você não possui acesso a este módulo.
        </div>
      )
    );
  }

  const ok = hasRequiredPermissions(
    permissions,
    requiredPermissions,
    requireAllPermissions,
  );

  if (!ok) {
    return (
      fallback ?? (
        <div className="rounded-xl border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-800">
          Você não possui permissão para acessar este conteúdo.
        </div>
      )
    );
  }

  return <>{children}</>;
}

// frontend/src/core/tenant/TenantContext.tsx

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type PropsWithChildren,
} from "react";

import { HttpError } from "../api/httpClient";
import { useAuth } from "../auth/useAuth";
import { isSuperAdmin } from "../auth/tipos";
import { getTenantBranding } from "./tenantApi";
import {
  buildFallbackTenantBranding,
  normalizeTenantBranding,
} from "./tenantBranding";
import type { TenantBranding, TenantContextValue } from "./tipos";

const TenantContext = createContext<TenantContextValue | undefined>(undefined);

export function TenantProvider({
  children,
}: PropsWithChildren) {
  const { user, isAuthenticated, isLoading: isAuthLoading } = useAuth();

  const [branding, setBranding] = useState<TenantBranding | null>(null);
  const [isLoading, setIsLoading] = useState(false);

  const refreshBranding = useCallback(async () => {
    if (
      !isAuthenticated ||
      !user ||
      isSuperAdmin(user) ||
      !user.locatarioId
    ) {
      setBranding(null);
      return;
    }

    setIsLoading(true);

    try {
      const response = await getTenantBranding(user.locatarioId);
      setBranding(normalizeTenantBranding(response.branding));
    } catch (error) {
      if (error instanceof HttpError) {
        setBranding(buildFallbackTenantBranding(user.locatarioId));
        setIsLoading(false);
        return;
      }

      setBranding(buildFallbackTenantBranding(user.locatarioId));
    } finally {
      setIsLoading(false);
    }
  }, [isAuthenticated, user]);

  useEffect(() => {
    if (isAuthLoading) {
      return;
    }

    void refreshBranding();
  }, [isAuthLoading, refreshBranding]);

  const value = useMemo<TenantContextValue>(
    () => ({
      branding,
      isLoading,
      refreshBranding,
    }),
    [branding, isLoading, refreshBranding],
  );

  return (
    <TenantContext.Provider value={value}>
      {children}
    </TenantContext.Provider>
  );
}

export function useTenant(): TenantContextValue {
  const context = useContext(TenantContext);

  if (!context) {
    throw new Error("useTenant deve ser usado dentro de TenantProvider");
  }

  return context;
}

// apps/web/src/features/superadmin/locatarios/hooks/useSalvarLocatario.ts

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { locatariosApi } from "../api/locatariosApi";
import type { AtualizarLocatarioInput, CriarLocatarioInput } from "../tipos";

type Params = {
  mode: "create" | "edit";
  locatarioId?: string;
  onSuccess?: (locatarioId?: string) => void | Promise<void>;
};

export function useSalvarLocatario({ mode, locatarioId, onSuccess, }: Params) {
  const [isSaving, setIsSaving] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  async function salvar(input: CriarLocatarioInput | AtualizarLocatarioInput) {
    setIsSaving(true);
    setErro(null);

    try {
      if (mode === "create") {
        const response = await locatariosApi.criar(input as CriarLocatarioInput);
        await onSuccess?.(response.item.locatarioId);
      } else {
        const id = locatarioId?.trim();

        if (!id) {
          setErro("ID do locatário inválido.");
          return false;
        }

        await locatariosApi.atualizar(id, input as AtualizarLocatarioInput);
        await onSuccess?.(id);
      }

      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {

===== apps/web/src/features/tenant-admin/assinatura/hooks/useAssinaturaModuloActions.ts =====

// apps/web/src/features/tenant-admin/assinatura/hooks/useAssinaturaModuloActions.ts

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { assinaturaModulosApi } from "../api/assinaturaModulosApi";

type ActionName = "vincular" | "desvincular";

type Params = {
  onSuccess?: () => void | Promise<void>;
};

export function useAssinaturaModuloActions({ onSuccess }: Params = {}) {
  const [isRunning, setIsRunning] = useState<ActionName | null>(null);
  const [erro, setErro] = useState<string | null>(null);
  const [sucesso, setSucesso] = useState<string | null>(null);

  async function vincular(moduloId: string) {
    const id = moduloId.trim();

    if (!id) {
      setErro("Informe o ID do módulo.");
      return false;
    }

    setIsRunning("vincular");
    setErro(null);
    setSucesso(null);

    try {
      await assinaturaModulosApi.vincular({ moduloId: id });
      setSucesso("Módulo vinculado com sucesso.");
      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsRunning(null);
    }
  }

  async function desvincular(moduloId: string) {
    const id = moduloId.trim();

    if (!id) {
      setErro("Módulo inválido.");
      return false;
    }

    setIsRunning("desvincular");
    setErro(null);
    setSucesso(null);

    try {
      await assinaturaModulosApi.desvincular(id);
      setSucesso("Módulo desvinculado com sucesso.");
      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsRunning(null);
    }
  }

  return {
    vincular,
    desvincular,
    isRunning,
    erro,
    sucesso,
  };
}

import { useCallback, useEffect, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { assinaturaModulosApi } from "../api/assinaturaModulosApi";
import type { ModuloAssinatura } from "../tipos";

export function useAssinaturaModulos() {
  const [locatarioId, setLocatarioId] = useState<string | null>(null);
  const [modulos, setModulos] = useState<ModuloAssinatura[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  const carregar = useCallback(async () => {
    setIsLoading(true);
    setErro(null);

    try {
      const response = await assinaturaModulosApi.listar();
      setLocatarioId(response.locatarioId);
      setModulos(response.modulos);
    } catch (error) {
      setModulos([]);
      setErro(mapError(error).message);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    void carregar();
  }, [carregar]);

  return {
    locatarioId,
    modulos,
    isLoading,
    erro,
    recarregar: carregar,
  };
}

// apps/web/src/features/tenant-admin/clientes/api/clientesApi.ts

import { httpClient } from "../../../../core/api/httpClient";
import type { AtualizarClienteInput, AtualizarClienteResponse, CriarClienteInput, CriarClienteResponse, ExcluirClienteResponse, ListarClientesResponse, ObterClienteResponse } from "../tipos";

const BASE = "/clientes";

export const clientesApi = {
  listar(): Promise<ListarClientesResponse> {
    return httpClient.get<ListarClientesResponse>(BASE);
  },

  obter(clienteId: string): Promise<ObterClienteResponse> {
    return httpClient.get<ObterClienteResponse>(`${BASE}/${clienteId}`);
  },

  criar(input: CriarClienteInput): Promise<CriarClienteResponse> {
    return httpClient.post<CriarClienteResponse, CriarClienteInput>(BASE, input);
  },

  atualizar(clienteId: string, input: AtualizarClienteInput): Promise<AtualizarClienteResponse> {
    return httpClient.patch<AtualizarClienteResponse, AtualizarClienteInput>(`${BASE}/${clienteId}`, input);
  },

  excluir(clienteId: string): Promise<ExcluirClienteResponse> {
    return httpClient.delete<ExcluirClienteResponse>(`${BASE}/${clienteId}`);
  },
};

  onSuccess?: () => void | Promise<void>;

  const [isSubmitting, setIsSubmitting] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

    } finally {
      setIsSubmitting(false);
    }
  }

    excluir,
    isSubmitting,
    erro,
  };

import { useEffect, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { clientesApi } from "../api/clientesApi";
import type { Cliente } from "../tipos";

export function useClienteDetails(clienteId?: string) {
  const [cliente, setCliente] = useState<Cliente | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  useEffect(() => {
    let ativo = true;

    async function carregar() {
      if (!clienteId?.trim()) {
        setErro("ID do cliente inválido.");
        setIsLoading(false);
        return;
      }

      setIsLoading(true);
      setErro(null);

      try {
        const response = await clientesApi.obter(clienteId);

        if (ativo) {
          setCliente(response.item);
        }
      } catch (error) {
        if (ativo) {
          setErro(mapError(error).message);
        }
      } finally {
        if (ativo) {
          setIsLoading(false);
        }
      }
    }

    void carregar();

    return () => {
      ativo = false;
    };
  }, [clienteId]);

  return {
    cliente,
    isLoading,
    erro,
  };
}

// apps/web/src/features/tenant-admin/clientes/hooks/useClientes.ts

import { useCallback, useEffect, useMemo, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { clientesApi } from "../api/clientesApi";
import type { Cliente } from "../tipos";

function filtrarClientes(clientes: Cliente[], busca: string): Cliente[] {
  const termo = busca.trim().toLowerCase();

  if (!termo) return clientes;

  return clientes.filter((cliente) =>
    cliente.nome.toLowerCase().includes(termo) ||
    cliente.documento?.toLowerCase().includes(termo) ||
    cliente.email?.toLowerCase().includes(termo),
  );
}

export function useClientes() {
  const [clientes, setClientes] = useState<Cliente[]>([]);
  const [busca, setBusca] = useState("");
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  const carregar = useCallback(async () => {
    setIsLoading(true);
    setErro(null);

    try {
      const response = await clientesApi.listar();
      setClientes(response.itens);
    } catch (error) {
      setErro(mapError(error).message);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    void carregar();
  }, [carregar]);

  const clientesFiltrados = useMemo(
    () => filtrarClientes(clientes, busca),
    [clientes, busca],
  );

  return {
    clientes,
    clientesFiltrados,
    busca,
    setBusca,
    isLoading,
    erro,
    recarregar: carregar,
  };
}

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { clientesApi } from "../api/clientesApi";
import type { AtualizarClienteInput, CriarClienteInput } from "../tipos";

type Params = {
  mode: "create" | "edit";
  clienteId?: string;
  onSuccess?: () => void | Promise<void>;
};

export function useSalvarCliente({ mode, clienteId, onSuccess }: Params) {
  const [isSaving, setIsSaving] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  async function salvar(input: CriarClienteInput | AtualizarClienteInput) {
    setIsSaving(true);
    setErro(null);

    try {
      if (mode === "create") {
        await clientesApi.criar(input as CriarClienteInput);
      } else {
        const id = clienteId?.trim();

        if (!id) {
          setErro("ID do cliente inválido.");
          return false;
        }

        await clientesApi.atualizar(id, input as AtualizarClienteInput);
      }

      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsSaving(false);
    }
  }

  return {
    salvar,
    isSaving,
    erro,
  };
}

// apps/web/src/features/tenant-admin/fornecedores/api/fornecedoresApi.ts

import { httpClient } from "../../../../core/api/httpClient";
import type { AtualizarFornecedorInput, AtualizarFornecedorResponse, CriarFornecedorInput, CriarFornecedorResponse, ExcluirFornecedorResponse, ListarFornecedoresResponse, ObterFornecedorResponse } from "../tipos";

const BASE = "/fornecedores";

export const fornecedoresApi = {
  listar(): Promise<ListarFornecedoresResponse> {
    return httpClient.get<ListarFornecedoresResponse>(BASE);
  },

  obter(fornecedorId: string): Promise<ObterFornecedorResponse> {
    return httpClient.get<ObterFornecedorResponse>(`${BASE}/${fornecedorId}`);
  },

  criar(input: CriarFornecedorInput): Promise<CriarFornecedorResponse> {
    return httpClient.post<CriarFornecedorResponse, CriarFornecedorInput>(BASE, input);
  },

  atualizar(fornecedorId: string, input: AtualizarFornecedorInput): Promise<AtualizarFornecedorResponse> {
    return httpClient.patch<AtualizarFornecedorResponse, AtualizarFornecedorInput>(`${BASE}/${fornecedorId}`, input);
  },

  excluir(fornecedorId: string): Promise<ExcluirFornecedorResponse> {
    return httpClient.delete<ExcluirFornecedorResponse>(`${BASE}/${fornecedorId}`);
  },
};

import { useEffect, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { fornecedoresApi } from "../api/fornecedoresApi";
import type { Fornecedor } from "../tipos";

export function useFornecedorDetails(fornecedorId?: string) {
  const [fornecedor, setFornecedor] = useState<Fornecedor | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  useEffect(() => {
    let ativo = true;

    async function carregar() {
      if (!fornecedorId?.trim()) {
        setErro("ID do fornecedor inválido.");
        setIsLoading(false);
        return;
      }

      setIsLoading(true);
      setErro(null);

      try {
        const response = await fornecedoresApi.obter(fornecedorId);

      if (ativo) {
        setFornecedor(response.item);
      }
      } catch (error) {
        if (ativo) {
          setErro(mapError(error).message);
        }
      } finally {
        if (ativo) {
          setIsLoading(false);
        }
      }
    }

    void carregar();

    return () => {
      ativo = false;
    };
  }, [fornecedorId]);

  return {
    fornecedor,
    isLoading,
    erro,
  };
}

===== apps/web/src/features/tenant-admin/fornecedores/hooks/useFornecedores.ts =====

// apps/web/src/features/tenant-admin/fornecedores/hooks/useFornecedores.ts

import { useCallback, useEffect, useMemo, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { fornecedoresApi } from "../api/fornecedoresApi";
import type { Fornecedor } from "../tipos";

function filtrarFornecedores(fornecedores: Fornecedor[], busca: string): Fornecedor[] {
  const termo = busca.trim().toLowerCase();

  if (!termo) return fornecedores;

  return fornecedores.filter((fornecedor) =>
    fornecedor.nome.toLowerCase().includes(termo) ||
    fornecedor.documento?.toLowerCase().includes(termo) ||
    fornecedor.email?.toLowerCase().includes(termo),
  );
}

export function useFornecedores() {
  const [fornecedores, setFornecedores] = useState<Fornecedor[]>([]);
  const [busca, setBusca] = useState("");
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  const carregar = useCallback(async () => {
    setIsLoading(true);
    setErro(null);

    try {
      const response = await fornecedoresApi.listar();
      setFornecedores(response.itens);
    } catch (error) {
      setErro(mapError(error).message);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    void carregar();
  }, [carregar]);

  const fornecedoresFiltrados = useMemo(
    () => filtrarFornecedores(fornecedores, busca),
    [fornecedores, busca],
  );

  return {
    fornecedores,
    fornecedoresFiltrados,
    busca,
    setBusca,
    isLoading,
    erro,
    recarregar: carregar,
  };
}

// apps/web/src/features/tenant-admin/fornecedores/hooks/useSalvarFornecedor.ts

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { fornecedoresApi } from "../api/fornecedoresApi";
import type { AtualizarFornecedorInput, CriarFornecedorInput } from "../tipos";

type Params = {
  mode: "create" | "edit";
  fornecedorId?: string;
  onSuccess?: () => void | Promise<void>;
};

export function useSalvarFornecedor({ mode, fornecedorId, onSuccess }: Params) {
  const [isSaving, setIsSaving] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  async function salvar(input: CriarFornecedorInput | AtualizarFornecedorInput) {
    setIsSaving(true);
    setErro(null);

    try {
      if (mode === "create") {
        await fornecedoresApi.criar(input as CriarFornecedorInput);
      } else {
        const id = fornecedorId?.trim();

        if (!id) {
          setErro("ID do fornecedor inválido.");
          return false;
        }

        await fornecedoresApi.atualizar(
          id,
          input as AtualizarFornecedorInput,
        );
      }

      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsSaving(false);
    }
  }

  return {
    salvar,
    isSaving,
    erro,
  };
}

import { useCallback, useEffect, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { meuLocatarioApi } from "../api/meuLocatarioApi";
import type { ConfiguracaoLocatario } from "../tipos";

export function useConfiguracoesLocatario(locatarioId?: string | null) {
  const [itens, setItens] = useState<ConfiguracaoLocatario[]>([]);
  const [configuracaoAtiva, setConfiguracaoAtiva] = useState<ConfiguracaoLocatario | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  const carregar = useCallback(async () => {
    setIsLoading(true);
    setErro(null);

    try {
      const [lista, ativa] = await Promise.all([
        meuLocatarioApi.listarConfiguracoes(),
        meuLocatarioApi.obterConfiguracaoAtiva(),

      setErro(mapError(error).message);

      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    void carregar();
  }, [carregar]);

  return {
    itens,
    configuracaoAtiva,
    isLoading,
    erro,
    recarregar: carregar,
  };

// apps/web/src/features/tenant-admin/locatario/hooks/useSalvarConfiguracaoLocatario.ts

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { meuLocatarioApi } from "../api/meuLocatarioApi";
import type { AtualizarConfiguracaoLocatarioInput, CriarConfiguracaoLocatarioInput } from "../tipos";

type Params = {
  mode: "create" | "edit";
  configuracaoId?: string;
  onSuccess?: () => void | Promise<void>;
};

export function useSalvarConfiguracaoLocatario({ mode, configuracaoId, onSuccess }: Params) {
  const [isSaving, setIsSaving] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  async function salvar(
    input:
      | CriarConfiguracaoLocatarioInput
      | AtualizarConfiguracaoLocatarioInput,
  ) {
    setIsSaving(true);
    setErro(null);

    try {
      if (mode === "create") {
        await meuLocatarioApi.criarConfiguracao(
          input as CriarConfiguracaoLocatarioInput,
        );
      } else {
        const confId = configuracaoId?.trim();

        if (!confId) {
          setErro("Configuração inválida.");
          return false;
        }

        await meuLocatarioApi.atualizarConfiguracao(
          confId,
          input as AtualizarConfiguracaoLocatarioInput,
        );
      }

      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsSaving(false);
    }
  }

  return {
    salvar,
    isSaving,
    erro,
  };
}

// apps/web/src/features/tenant-admin/produtos/api/produtosApi.ts

import { httpClient } from "../../../../core/api/httpClient";
import type { AtualizarProdutoInput, AtualizarProdutoResponse, CriarProdutoInput, CriarProdutoResponse, ExcluirProdutoResponse, ListarProdutosResponse, ObterProdutoResponse } from "../tipos";

const BASE = "/produtos";

export const produtosApi = {
  listar(): Promise<ListarProdutosResponse> {
    return httpClient.get<ListarProdutosResponse>(BASE);
  },

  obter(produtoId: string): Promise<ObterProdutoResponse> {
    return httpClient.get<ObterProdutoResponse>(`${BASE}/${produtoId}`);
  },

  criar(input: CriarProdutoInput): Promise<CriarProdutoResponse> {
    return httpClient.post<CriarProdutoResponse, CriarProdutoInput>(BASE, input);
  },

  atualizar(produtoId: string, input: AtualizarProdutoInput): Promise<AtualizarProdutoResponse> {
    return httpClient.patch<AtualizarProdutoResponse, AtualizarProdutoInput>(`${BASE}/${produtoId}`, input);
  },

  excluir(produtoId: string): Promise<ExcluirProdutoResponse> {
    return httpClient.delete<ExcluirProdutoResponse>(`${BASE}/${produtoId}`);
  },
};

function parsePreco(value: string): number {
  const normalized = value.trim().replace(",", ".");
  const parsed = Number(normalized);

  return Number.isFinite(parsed) ? parsed : 0;
}

export function ProdutoForm({ mode, initialData, isSaving, erro, onSubmit }: Props) {

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { produtosApi } from "../api/produtosApi";

type Params = {
  onSuccess?: () => void | Promise<void>;
};

export function useProdutoActions({ onSuccess }: Params = {}) {
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  async function excluir(produtoId: string) {
    const id = produtoId.trim();

    if (!id) {
      setErro("ID do produto inválido.");
      return false;
    }

    setIsSubmitting(true);
    setErro(null);

    try {
      await produtosApi.excluir(id);
      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsSubmitting(false);
    }
  }

  return {
    excluir,
    isSubmitting,
    erro,
  };
}

import { mapError } from "../../../../core/errors/mapError";
import { produtosApi } from "../api/produtosApi";

export function useProdutoDetails(produtoId?: string) {
  const [produto, setProduto] = useState<Produto | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  useEffect(() => {
    let ativo = true;

    async function carregar() {
      if (!produtoId?.trim()) {
        setErro("ID do produto inválido.");
        setIsLoading(false);
        return;
      }

      setIsLoading(true);
      setErro(null);

      try {
        const response = await produtosApi.obter(produtoId);

        if (ativo) {
          setProduto(response);
        }
      } catch (error) {
        if (ativo) {
          setErro(mapError(error).message);
        }
      } finally {
        if (ativo) {
          setIsLoading(false);
        }
      }
    }

    void carregar();

    return () => {
      ativo = false;
    };
  }, [produtoId]);

  return {
    produto,
    isLoading,
    erro,
  };
}

// apps/web/src/features/tenant-admin/produtos/hooks/useProdutos.ts

import { useCallback, useEffect, useMemo, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { produtosApi } from "../api/produtosApi";
import type { Produto, TipoProduto } from "../tipos";

type FiltroTipoProduto = TipoProduto | "TODOS";

function filtrarProdutos(produtos: Produto[], busca: string, tipo: FiltroTipoProduto): Produto[] {
  const termo = busca.trim().toLowerCase();

  return produtos.filter((produto) => {
    const atendeBusca =
      !termo ||
      produto.nome.toLowerCase().includes(termo) ||
      produto.descricao?.toLowerCase().includes(termo);

    const atendeTipo = tipo === "TODOS" || produto.tipo === tipo;

    return atendeBusca && atendeTipo;
  });
}

export function useProdutos() {
  const [produtos, setProdutos] = useState<Produto[]>([]);
  const [busca, setBusca] = useState("");
  const [tipo, setTipo] = useState<FiltroTipoProduto>("TODOS");
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  const carregar = useCallback(async () => {
    setIsLoading(true);
    setErro(null);

    try {
      const response = await produtosApi.listar();
      setProdutos(response.itens);
    } catch (error) {
      setErro(mapError(error).message);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    void carregar();
  }, [carregar]);

  const produtosFiltrados = useMemo(
    () => filtrarProdutos(produtos, busca, tipo),
    [produtos, busca, tipo],
  );

  return {
    produtos,
    produtosFiltrados,
    busca,
    setBusca,
    tipo,
    setTipo,
    isLoading,
    erro,
    recarregar: carregar,
  };
}

// apps/web/src/features/tenant-admin/produtos/hooks/useSalvarProduto.ts

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { produtosApi } from "../api/produtosApi";
import type { AtualizarProdutoInput, CriarProdutoInput } from "../tipos";

type Params = {
  mode: "create" | "edit";
  produtoId?: string;
  onSuccess?: () => void | Promise<void>;
};

export function useSalvarProduto({ mode, produtoId, onSuccess }: Params) {
  const [isSaving, setIsSaving] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  async function salvar(input: CriarProdutoInput | AtualizarProdutoInput) {
    setIsSaving(true);
    setErro(null);

    try {
      if (mode === "create") {
        await produtosApi.criar(input as CriarProdutoInput);
      } else {
        const id = produtoId?.trim();

        if (!id) {
          setErro("ID do produto inválido.");
          return false;
        }

        await produtosApi.atualizar(id, input as AtualizarProdutoInput);
      }

      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsSaving(false);
    }
  }

  return {
    salvar,
    isSaving,
    erro,
  };
}

import { httpClient } from "../../../../core/api/httpClient";
import type { AdicionarItemTransacaoInput, AdicionarItemTransacaoResponse, AtualizarItemTransacaoInput, AtualizarItemTransacaoResponse, CancelarTransacaoInput, CancelarTransacaoResponse,
  ConfirmarTransacaoResponse, CriarCompraInput, CriarTransacaoResponse, CriarVendaInput, ListarTransacoesResponse, ObterTransacaoResponse, RemoverItemTransacaoResponse,
} from "../tipos";

const BASE = "/transacoes";

export const transacoesApi = {
  listar(): Promise<ListarTransacoesResponse> {
    return httpClient.get<ListarTransacoesResponse>(BASE);
  },

  obter(transacaoId: string): Promise<ObterTransacaoResponse> {
    return httpClient.get<ObterTransacaoResponse>(`${BASE}/${transacaoId}`);
  },

  criarVenda(input: CriarVendaInput): Promise<CriarTransacaoResponse> {
    return httpClient.post<CriarTransacaoResponse, CriarVendaInput>("/vendas", input);
  },

  criarCompra(input: CriarCompraInput): Promise<CriarTransacaoResponse> {
    return httpClient.post<CriarTransacaoResponse, CriarCompraInput>("/compras", input);
  },

  adicionarItem(transacaoId: string, input: AdicionarItemTransacaoInput): Promise<AdicionarItemTransacaoResponse> {
    return httpClient.post<AdicionarItemTransacaoResponse, AdicionarItemTransacaoInput>(`${BASE}/${transacaoId}/itens`, input);
  },

  atualizarItem(transacaoId: string, itemId: string, input: AtualizarItemTransacaoInput): Promise<AtualizarItemTransacaoResponse> {
    return httpClient.patch<AtualizarItemTransacaoResponse, AtualizarItemTransacaoInput>(`${BASE}/${transacaoId}/itens/${itemId}`, input);
  },

  removerItem(transacaoId: string, itemId: string): Promise<RemoverItemTransacaoResponse> {
    return httpClient.delete<RemoverItemTransacaoResponse>(`${BASE}/${transacaoId}/itens/${itemId}`);
  },

  confirmar(transacaoId: string): Promise<ConfirmarTransacaoResponse> {
    return httpClient.post<ConfirmarTransacaoResponse>(`${BASE}/${transacaoId}/confirmar`);
  },

  cancelar(transacaoId: string, input: CancelarTransacaoInput = {}): Promise<CancelarTransacaoResponse> {
    return httpClient.post<CancelarTransacaoResponse, CancelarTransacaoInput>(`${BASE}/${transacaoId}/cancelar`, input);
  }
};

===== apps/web/src/features/tenant-admin/transacoes/hooks/useTransacaoActions.ts =====

// apps/web/src/features/tenant-admin/transacoes/hooks/useTransacaoActions.ts

import { useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { transacoesApi } from "../api/transacoesApi";
import type { AdicionarItemTransacaoInput, AtualizarItemTransacaoInput, CancelarTransacaoInput } from "../tipos";

type Params = {
  transacaoId?: string;
  onSuccess?: () => void | Promise<void>;
};

export function useTransacaoActions({ transacaoId, onSuccess }: Params = {}) {
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  function validarId(): string | null {
    const id = transacaoId?.trim();

    if (!id) {
      setErro("ID da transação inválido.");
      return null;
    }

    return id;
  }

  async function executar(fn: (id: string) => Promise<unknown>) {
    const id = validarId();
    if (!id) return false;

    setIsSubmitting(true);
    setErro(null);

    try {
      await fn(id);
      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsSubmitting(false);
    }
  }

  return {
    isSubmitting,
    erro,

    adicionarItem(input: AdicionarItemTransacaoInput) {
      return executar((id) => transacoesApi.adicionarItem(id, input));
    },

    atualizarItem(itemId: string, input: AtualizarItemTransacaoInput) {
      return executar((id) => transacoesApi.atualizarItem(id, itemId, input));
    },

    removerItem(itemId: string) {
      return executar((id) => transacoesApi.removerItem(id, itemId));
    },

    confirmar() {
      return executar((id) => transacoesApi.confirmar(id));
    },

    cancelar(input: CancelarTransacaoInput = {}) {
      return executar((id) => transacoesApi.cancelar(id, input));
    },
  };
}

export function useTransacaoDetails(transacaoId?: string) {
  const [transacao, setTransacao] = useState<Transacao | null>(null);
  const [isLoading, setIsLoading] = useState(true);

// apps/web/src/features/tenant-admin/transacoes/hooks/useTransacoes.ts

import { useCallback, useEffect, useMemo, useState } from "react";
import { mapError } from "../../../../core/errors/mapError";
import { transacoesApi } from "../api/transacoesApi";
import type { TipoTransacao, Transacao } from "../tipos";

type FiltroTipo = TipoTransacao | "TODOS";

function filtrarTransacoes(
  transacoes: Transacao[],
  busca: string,
  tipo: FiltroTipo,
): Transacao[] {
  const termo = busca.trim().toLowerCase();

  return transacoes.filter((transacao) => {
    const atendeTipo = tipo === "TODOS" || transacao.tipo === tipo;
    const atendeBusca =
      !termo ||
      transacao.transacaoId.toLowerCase().includes(termo) ||
      transacao.status.toLowerCase().includes(termo);

    return atendeTipo && atendeBusca;
  });
}

export function useTransacoes() {
  const [transacoes, setTransacoes] = useState<Transacao[]>([]);
  const [busca, setBusca] = useState("");
  const [tipo, setTipo] = useState<FiltroTipo>("TODOS");
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);

  const carregar = useCallback(async () => {
    setIsLoading(true);
    setErro(null);

    try {
      const response = await transacoesApi.listar();
      setTransacoes(response.itens);
    } catch (error) {
      setErro(mapError(error).message);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    void carregar();
  }, [carregar]);

  const transacoesFiltradas = useMemo(
    () => filtrarTransacoes(transacoes, busca, tipo),
    [transacoes, busca, tipo],
  );

  return {
    transacoes,
    transacoesFiltradas,
    busca,
    setBusca,
    tipo,
    setTipo,
    isLoading,
    erro,
    recarregar: carregar,
  };
}

import type { AlterarMinhaSenhaInput, RedefinirSenhaUsuarioInput } from "../tipos";

type Props = {
  mode: "alterar" | "redefinir";
  isSaving?: boolean;
  erro?: string | null;
  sucesso?: string | null;
  onSubmit: (
    input: AlterarMinhaSenhaInput | RedefinirSenhaUsuarioInput,
  ) => void | Promise<void>;
};

export function UsuarioSenhaForm({ mode, isSaving, erro, sucesso, onSubmit }: Props) {
  const [senhaAtual, setSenhaAtual] = useState("");
  const [novaSenha, setNovaSenha] = useState("");
  const [confirmacao, setConfirmacao] = useState("");

  const titulo = useMemo(
    () =>
      mode === "alterar" ? "Alterar minha senha" : "Redefinir senha do usuário",
    [mode],
  );

  async function handleSubmit(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (novaSenha !== confirmacao) {
      return;
    }

    if (mode === "alterar") {
      await onSubmit({

        novaSenha,
      });
    }

          {erro}
        </div>
      ) : null}

      {sucesso ? (
        <div className="rounded-xl border border-green-200 bg-green-50 px-3 py-2 text-sm text-green-700">
          {sucesso}
        </div>
      ) : null}

            className="rounded-xl border px-3 py-2"

            onChange={(event) => setSenhaAtual(event.target.value)}
            minLength={4}
            required
          />
        </label>
      ) : null}

      <label className="grid gap-1">
        <span className="text-sm font-medium">Nova senha</span>
        <input
          className="rounded-xl border px-3 py-2"
          type="password"
          value={novaSenha}
          onChange={(event) => setNovaSenha(event.target.value)}
          minLength={4}
          required
        />
      </label>

      <label className="grid gap-1">
        <span className="text-sm font-medium">Confirmar nova senha</span>
        <input
          className="rounded-xl border px-3 py-2"

          value={confirmacao}
          onChange={(event) => setConfirmacao(event.target.value)}

                        to={`/app/usuarios/${usuario.usuarioId}/editar`}
                      >
                        Editar
                      </Link>

                      {usuario.status === "ATIVO" ? (
                        <button
                          className="text-red-600 underline"
                          type="button"
                          onClick={() => onInativar?.(usuario.usuarioId)}
                        >
                          Inativar
                        </button>
                      ) : null}

                      {usuario.status === "INATIVO" ? (
                        <button
                          className="text-green-600 underline"
                          type="button"
                          onClick={() => onReativar?.(usuario.usuarioId)}
                        >
                          Reativar
                        </button>
                      ) : null}
                    </>
                  ) : (
                    <span className="text-xs text-slate-500">
                      Sem permissão de alteração
                    </span>
                  )}
                </div>
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

import { useState } from "react";
import { AtualizarUsuarioInput, CriarUsuarioInput, UsuarioFormMode, usuariosApi } from "..";
import { mapError } from "../../../../core/errors/mapError";

/**
 * 
 */
type Params = {
  mode: UsuarioFormMode;
  usuarioId?: string;
  onSuccess?: () => void | Promise<void>;
};

/**
 * 
 * @param param0 
 * @returns 
 */
export function useSalvarUsuario({ mode, usuarioId, onSuccess }: Params) {
  const [isSaving, setIsSaving] = useState(false);
  const [erro, setErro] = useState<string | null>(null);

  /**
   * 
   * @param input 
   * @returns 
   */
  async function salvar(input: CriarUsuarioInput | AtualizarUsuarioInput) {
    setIsSaving(true);
    setErro(null);

    try {
      if (mode === "create") {
        await usuariosApi.criar(input as CriarUsuarioInput);
      } else {
        const usuarioIdNormalizado = usuarioId?.trim();

        if (!usuarioIdNormalizado) {
          setErro("ID do usuário inválido.");
          return false;
        }

        await usuariosApi.atualizar(
          usuarioIdNormalizado,
          input as AtualizarUsuarioInput,
        );
      }

      await onSuccess?.();
      return true;
    } catch (error) {
      setErro(mapError(error).message);
      return false;
    } finally {
      setIsSaving(false);
    }
  }

  return {
    salvar,
    isSaving,
    erro,
  };
}

===== apps/web/src/features/tenant-admin/usuarios/hooks/useUsuarios.ts =====

// web/src/features/tenant-admin/usuarios/hooks/useUsuarios.ts

import { useCallback, useEffect, useMemo, useState } from "react";
import { FiltroUsuarios, Usuario, usuariosApi } from "..";
import { mapError } from "../../../../core/errors/mapError";


/**
 * 
 * @param usuarios 
 * @param filtro 
 * @returns 
 */
function filtrarUsuarios(usuarios: Usuario[], filtro: FiltroUsuarios): Usuario[] {
  const busca = filtro.busca?.trim().toLowerCase() ?? "";
  const status = filtro.status ?? "TODOS";

  return usuarios.filter((usuario) => {
    const atendeBusca =
      busca.length === 0 ||
      usuario.nome.toLowerCase().includes(busca) ||
      usuario.email.toLowerCase().includes(busca);

    const atendeStatus = status === "TODOS" || usuario.status === status;
    return atendeBusca && atendeStatus;
  });
}

/**
 * 
 * @returns 
 */
export function useUsuarios() {
  const [usuarios, setUsuarios] = useState<Usuario[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [erro, setErro] = useState<string | null>(null);
  const [filtro, setFiltro] = useState<FiltroUsuarios>({
    busca: "",
    status: "TODOS",
  });

  const carregar = useCallback(async () => {
    setIsLoading(true);
    setErro(null);

    try {
      const response = await usuariosApi.listar();
      setUsuarios(response.itens);
    } catch (error) {
      setErro(mapError(error).message);
    } finally {
      setIsLoading(false);
    }
  }, []);

  useEffect(() => {
    void carregar();
  }, [carregar]);

  const usuariosFiltrados = useMemo(
    () => filtrarUsuarios(usuarios, filtro),
    [usuarios, filtro],
  );

  return {
    usuarios,
    usuariosFiltrados,
    isLoading,
    erro,
    filtro,
    setFiltro,
    recarregar: carregar,
  };
}

export default function UsuarioDetailsPage() {
  const params = useParams<{ usuarioId: string }>();
  const usuarioId = params.usuarioId?.trim();

  const { usuario, isLoading, erro } = useUsuarioDetails(usuarioId);

  const {
    redefinirSenha,
    erro: erroSenha,
    sucesso: sucessoSenha,
    isSaving: isSavingSenha,
  } = useSenhaUsuario({
    usuarioId,
  });

  async function handleRedefinirSenha(input: RedefinirSenhaUsuarioInput) {
    await redefinirSenha(input);
  }

  if (isLoading) {
    return <div className="rounded-2xl border bg-white p-6">Carregando usuário...</div>;
  }

  if (erro) {
    return (
      <div className="rounded-xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700">
        {erro}
      </div>
    );
  }

  if (!usuario) {
    return (
      <div className="rounded-xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700">
        Usuário não encontrado.
      </div>
    );
  }

  return (
    <div className="grid gap-6">
      <div className="flex items-start justify-between gap-4">
        <div>
          <h1 className="text-2xl font-semibold">{usuario.nome}</h1>
          <p className="text-sm text-slate-600">{usuario.email}</p>
        </div>

        <PageGuard
          requiredModule={USUARIOS_MODULE}
          requiredPermissions={[USUARIOS_PERMISSIONS.ATUALIZAR]}
          fallback={
            <div className="rounded-xl border border-amber-200 bg-amber-50 px-3 py-2 text-sm text-amber-800">
              Você pode visualizar, mas não editar este usuário.
            </div>
          }
        >
          <Link
            className="rounded-xl border px-4 py-2"
            to={`/app/usuarios/${usuario.usuarioId}/editar`}
          >
            Editar
          </Link>
        </PageGuard>
      </div>

      <div className="grid gap-3 rounded-2xl border bg-white p-6 text-sm">
        <div><strong>ID:</strong> {usuario.usuarioId}</div>
        <div><strong>Nome:</strong> {usuario.nome}</div>
        <div><strong>E-mail:</strong> {usuario.email}</div>
        <div><strong>Status:</strong> {usuario.status}</div>
        <div><strong>Último login:</strong> {usuario.ultimoLoginEm ?? "-"}</div>
        <div><strong>Criado em:</strong> {usuario.criadoEm}</div>
        <div><strong>Atualizado em:</strong> {usuario.atualizadoEm}</div>
      </div>

      <PageGuard
        requiredModule={USUARIOS_MODULE}
        requiredPermissions={[USUARIOS_PERMISSIONS.ATUALIZAR]}
        fallback={
          <div className="rounded-xl border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-800">
            Você não possui permissão para redefinir senha.
          </div>
        }
      >
        <UsuarioSenhaForm
          mode="redefinir"
          isSaving={isSavingSenha}

// apps/web/src/features/tenant-admin/vendas/pages/NovaVendaPage.tsx

import { useState } from "react";
import { useNavigate } from "react-router-dom";
import { mapError } from "../../../../core/errors/mapError";
import { vendasApi } from "../api/vendasApi";

export default function NovaVendaPage() {
  const navigate = useNavigate();

  const [clienteId, setClienteId] = useState("");
  const [observacao, setObservacao] = useState("");
  const [erro, setErro] = useState<string | null>(null);
  const [isSaving, setIsSaving] = useState(false);

  async function handleSubmit(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault();

    setErro(null);
    setIsSaving(true);

    try {
      const response = await vendasApi.criar({
        clienteId: clienteId.trim() || null,
        observacao: observacao.trim() || null,
      });

      navigate(`/app/transacoes/${response.item.transacaoId}`);
    } catch (error) {
      setErro(mapError(error).message);
    } finally {
      setIsSaving(false);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <h1>Nova venda</h1>

      <p>Crie uma venda para depois adicionar os itens.</p>

      {erro ? <p role="alert">{erro}</p> : null}

      <label>
        Cliente ID opcional
        <input
          value={clienteId}
          onChange={(event) => setClienteId(event.target.value)}
          placeholder="Cliente ID"
        />
      </label>

      <label>
        Observação
        <textarea
          value={observacao}
          onChange={(event) => setObservacao(event.target.value)}
          placeholder="Observação da venda"
        />
      </label>

      <button type="submit" disabled={isSaving}>
        {isSaving ? "Criando..." : "Criar venda"}
      </button>
    </form>
  );
}
