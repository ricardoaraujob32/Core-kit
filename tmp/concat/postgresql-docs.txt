===== docs/banco-de-dados/postgresql/ddl/create_roles_login.sql =====

-- ============================================================
-- Padrao multilocatario: Roles-modelo (NOLOGIN) reutilizáveis
-- Execute como: postgres (superuser) ou role com CREATEROLE
-- ============================================================

DO $$
BEGIN
  -- 1) administrador (owner lógico): ideal para ser OWNER de schemas/objetos
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'multilocatario_administrador') THEN
    CREATE ROLE multilocatario_administrador LOGIN PASSWORD '1234';
  END IF;

  -- 2) Migrador: usado por migrations (DDL controlado)
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'multilocatario_migrador') THEN
    CREATE ROLE multilocatario_migrador LOGIN PASSWORD '1234';
  END IF;

  -- 3) leitor: SELECT
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'multilocatario_leitor') THEN
    CREATE ROLE multilocatario_leitor LOGIN PASSWORD '1234';
  END IF;

  -- 4) aplicacao: INSERT/UPDATE/DELETE (geralmente inclui leitura também)
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'multilocatario_aplicacao') THEN
    CREATE ROLE multilocatario_aplicacao LOGIN PASSWORD '1234';
  END IF;

  -- 5) Auditor: leitura de trilhas, visões de auditoria, etc
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'multilocatario_auditor') THEN
    CREATE ROLE multilocatario_auditor LOGIN PASSWORD '1234';
  END IF;

  -- 6) Joperador_job: rotinas agendadas/background (ETL, import/export, etc)
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'multilocatario_operador_job') THEN
    CREATE ROLE multilocatario_operador_job LOGIN PASSWORD '1234';
  END IF;
END $$;


===== docs/banco-de-dados/postgresql/ddl/create_roles_nologin.sql =====

-- ============================================================
-- Padrao CoreKit: Roles-modelo (NOLOGIN) reutilizáveis
-- Execute como: postgres (superuser) ou role com CREATEROLE
-- ============================================================

DO $$
BEGIN
  -- 1) posse (owner lógico): ideal para ser OWNER de schemas/objetos
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'posse') THEN
    CREATE ROLE posse NOLOGIN;
  END IF;

  -- 2) migração: usado por migrations (DDL controlado)
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'migracao') THEN
    CREATE ROLE migracao NOLOGIN;
  END IF;

  -- 3) Leitura: SELECT
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'leitura') THEN
    CREATE ROLE leitura NOLOGIN;
  END IF;

  -- 4) Escrita: INSERT/UPDATE/DELETE (geralmente inclui leitura também)
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'escrita') THEN
    CREATE ROLE escrita NOLOGIN;
  END IF;

  -- 5) auditoria: leitura de trilhas, visões de auditoria, etc
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'auditoria') THEN
    CREATE ROLE auditoria NOLOGIN;
  END IF;

  -- 6) automacao: rotinas agendadas/background (ETL, import/export, etc)
  IF NOT EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'automacao') THEN
    CREATE ROLE automacao NOLOGIN;
  END IF;
END $$;

REVOKE ALL ON SCHEMA public FROM PUBLIC;

-- 2) Criar schemas do Core Kit

CREATE SCHEMA IF NOT EXISTS audit AUTHORIZATION posse;
CREATE SCHEMA IF NOT EXISTS jobs AUTHORIZATION posse;
CREATE SCHEMA IF NOT EXISTS api AUTHORIZATION posse;

GRANT ALL ON SCHEMA main TO migracao;
GRANT ALL ON SCHEMA api TO migracao;
GRANT ALL ON SCHEMA audit TO migracao;
GRANT ALL ON SCHEMA jobs TO migracao;

-- 3) Acesso mínimo para roles (USAGE permite "enxergar" objetos no schema)

ALTER DEFAULT PRIVILEGES IN SCHEMA api GRANT SELECT ON SEQUENCES TO leitura;

-- db/bootstrap/02_privileges.sql
-- Finalidade: definir privilégios mínimos e default privileges (essencial para robustez).



-- Exemplo: criar logins e associar às roles-modelo
CREATE ROLE ck_app LOGIN PASSWORD '...';
GRANT corekit_escrita TO ck_app;

CREATE ROLE ck_migrate LOGIN PASSWORD '...';
GRANT corekit_migrador TO ck_migrate;

CREATE ROLE ck_ro LOGIN PASSWORD '...';
GRANT corekit_leitura TO ck_ro;

CREATE ROLE ck_job LOGIN PASSWORD '...';
GRANT corekit_job TO ck_job;


-- 1) Banco: remover privilégios amplos do PUBLIC (no banco atual)
-- (Rodar conectado no banco do projeto)
REVOKE ALL ON DATABASE current_database() FROM PUBLIC;

-- 2) Permitir que roles conectem no banco (ajuste se quiser bloquear leitura/auditor por ambiente)
GRANT CONNECT ON DATABASE current_database() TO corekit_migrador, corekit_app, corekit_leitura, corekit_auditor;

-- 3) Permissões em tabelas existentes (caso já existam objetos)
-- APP schema: runtime pode DML, leitura só SELECT
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA sch_corekit_app TO corekit_app;
GRANT SELECT ON ALL TABLES IN SCHEMA sch_corekit_app TO corekit_leitura;

-- JOBS schema: runtime pode DML (sch_corekit_jobs de import/export), leitura só SELECT
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA sch_corekit_jobs TO corekit_app;
GRANT SELECT ON ALL TABLES IN SCHEMA sch_corekit_jobs TO corekit_leitura;

-- AUDIT schema: auditor pode SELECT (e opcional INSERT se auditar pelo BD)
GRANT SELECT ON ALL TABLES IN SCHEMA sch_corekit_audit TO corekit_auditor;
GRANT SELECT ON ALL TABLES IN SCHEMA sch_corekit_audit TO corekit_leitura;

-- Opcional: permitir a aplicação ler auditoria
-- GRANT SELECT ON ALL TABLES IN SCHEMA sch_corekit_audit TO corekit_app;

-- 4) Sequences (se usar SERIAL/IDENTITY/sequence)
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA sch_corekit_app TO corekit_app;
GRANT USAGE, SELECT ON ALL SEQUENCES IN SCHEMA sch_corekit_jobs TO corekit_app;

GRANT SELECT ON ALL SEQUENCES IN SCHEMA sch_corekit_app TO corekit_leitura;
GRANT SELECT ON ALL SEQUENCES IN SCHEMA sch_corekit_jobs TO corekit_leitura;

-- 5) DEFAULT PRIVILEGES — para objetos futuros criados pelo migrador
-- Importante: DEFAULT PRIVILEGES são POR "criador" + schema.
-- Aqui assumimos que migrations rodam como corekit_migrador_dev/hom/prod,
-- e os objetos ficam "criadas por" corekit_migrador (via IN ROLE).
--
-- Dica: rode migrations sempre usando a credencial migradora.

-- Para tabelas novas no schema sch_corekit_app
ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_app
  GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO corekit_app;

ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_app
  GRANT SELECT ON TABLES TO corekit_leitura;

-- Para sequences novas no schema sch_corekit_app
ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_app
  GRANT USAGE, SELECT ON SEQUENCES TO corekit_app;

ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_app
  GRANT SELECT ON SEQUENCES TO corekit_leitura;

-- Para tabelas novas no schema sch_corekit_jobs
ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_jobs
  GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO corekit_app;

ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_jobs
  GRANT SELECT ON TABLES TO corekit_leitura;

-- Para sequences novas no schema sch_corekit_jobs
ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_jobs
  GRANT USAGE, SELECT ON SEQUENCES TO corekit_app;

ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_jobs
  GRANT SELECT ON SEQUENCES TO corekit_leitura;

-- Para tabelas novas no schema sch_corekit_audit
ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_audit
  GRANT SELECT ON TABLES TO corekit_auditor;

ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_audit
  GRANT SELECT ON TABLES TO corekit_leitura;

-- Opcional: permitir INSERT em sch_corekit_audit se você decidir auditar via triggers no BD
-- ALTER DEFAULT PRIVILEGES FOR ROLE corekit_migrador IN SCHEMA sch_corekit_audit
--   GRANT INSERT ON TABLES TO corekit_auditor;

-- 6) Hardening opcional: impedir CREATE no schema (somente migrador deve criar)
REVOKE CREATE ON SCHEMA sch_corekit_app FROM PUBLIC;
REVOKE CREATE ON SCHEMA sch_corekit_jobs FROM PUBLIC;

GRANT posse, migracao, escrita, leitura TO multilocatario_administrador;

-- app (principal)

-- api (secundárias)
GRANT leitura TO multilocatario_leitor;

-- auditor
GRANT auditoria TO multilocatario_auditor;

-- job
GRANT automacao TO multilocatario_operador_job;

-- leitura enxerga schemas
GRANT USAGE ON SCHEMA core TO corekit_leitura;
GRANT USAGE ON SCHEMA jobs TO corekit_leitura;

-- escrita enxerga schemas (via herança) e faz DML (não precisa CREATE)
GRANT USAGE ON SCHEMA core TO corekit_escrita;
GRANT USAGE ON SCHEMA jobs TO corekit_escrita;

-- auditor enxerga auditoria
GRANT USAGE ON SCHEMA auditoria TO corekit_auditor;

-- job enxerga jobs (e opcionalmente core)
GRANT USAGE ON SCHEMA jobs TO corekit_job;
GRANT USAGE ON SCHEMA core TO corekit_job; -- opcional, se job precisar ler core

-- migrador pode criar objetos
GRANT USAGE, CREATE ON SCHEMA core TO corekit_migrador;
GRANT USAGE, CREATE ON SCHEMA jobs TO corekit_migrador;
GRANT USAGE, CREATE ON SCHEMA auditoria TO corekit_migrador;

-- hardening: bloquear CREATE do PUBLIC
REVOKE CREATE ON SCHEMA core FROM PUBLIC;
REVOKE CREATE ON SCHEMA jobs FROM PUBLIC;
REVOKE CREATE ON SCHEMA auditoria FROM PUBLIC;

===== docs/banco-de-dados/postgresql/funcoes/auditar_evento.sql =====

-- =========================================================
-- util: auditoria (genérica)
-- =========================================================
CREATE OR REPLACE FUNCTION fn_auditar_evento(
  p_tipo_evento text,
  p_acao text,
  p_recurso text,
  p_recurso_id text,
  p_resultado text,
  p_detalhes jsonb
)
RETURNS void AS $$
DECLARE
  v_usuario_id uuid;
  v_locatario_id uuid;
  v_ip text;
  v_user_agent text;
  v_corr text;
BEGIN
  v_usuario_id := NULLIF(fn_configuracoes_app('app.usuario_id'), '')::uuid;
  v_locatario_id := NULLIF(fn_configuracoes_app('app.locatario_id'), '')::uuid;
  v_ip := fn_configuracoes_app('app.ip');
  v_user_agent := fn_configuracoes_app('app.user_agent');
  v_corr := fn_configuracoes_app('app.correlation_id');

  INSERT INTO auditoria (
    id,
    locatario_id,
    usuario_id,
    tipo_evento,
    acao,
    recurso,
    recurso_id,
    resultado,
    detalhes,
    ip,
    user_agent,
    correlation_id,
    criado_em
  ) VALUES (
    gen_random_uuid(),
    v_locatario_id,
    v_usuario_id,
    p_tipo_evento,
    p_acao,
    p_recurso,
    p_recurso_id,
    p_resultado,
    COALESCE(p_detalhes, '{}'::jsonb),
    v_ip,
    v_user_agent,
    v_corr,
    now()
  );
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION app_definir_contexto(p_usuario_id uuid, p_locatario_id uuid, p_is_plataforma boolean)
RETURNS void
LANGUAGE plpgsql
AS $$
DECLARE
  user_locatario uuid;
BEGIN
  SELECT u.locatario_id INTO user_locatario
  FROM usuarios u
  WHERE u.usuario_id = p_usuario_id
    AND u.excluido_em IS NULL;

  IF user_locatario IS NULL THEN
    -- Usuário de plataforma
    IF NOT p_is_plataforma THEN
      RAISE EXCEPTION 'usuário plataforma deve usar p_is_plataforma=true';
    END IF;

    -- Pode setar qualquer locatário alvo (inclusive NULL para modo plataforma puro)
    PERFORM set_config('app.usuario_id', p_usuario_id::text, true);
    PERFORM set_config('app.is_plataforma', 'true', true);

    IF p_locatario_id IS NULL THEN
      PERFORM set_config('app.locatario_id', '', true);
    ELSE
      PERFORM set_config('app.locatario_id', p_locatario_id::text, true);
    END IF;

  ELSE
    -- Usuário comum: só pode atuar no próprio locatário
    IF p_locatario_id IS DISTINCT FROM user_locatario THEN
      RAISE EXCEPTION 'usuário comum só pode atuar no próprio locatário';
    END IF;

    PERFORM set_config('app.usuario_id', p_usuario_id::text, true);
    PERFORM set_config('app.is_plataforma', 'false', true);
    PERFORM set_config('app.locatario_id', user_locatario::text, true);
  END IF;
END;
$$;



-- Usuário comum:
SELECT app_definir_contexto(:usuario_id, :locatario_id_do_usuario, false);

-- Superadmin, modo plataforma:
SELECT app_definir_contexto(:usuario_id, NULL, true);

-- Superadmin, atuando num locatário:
SELECT app_definir_contexto(:usuario_id, :locatario_alvo_id, true);

CREATE OR REPLACE FUNCTION app_usuario_id()
RETURNS uuid
LANGUAGE sql STABLE
AS $$
  SELECT NULLIF(current_setting('app.usuario_id', true), '')::uuid
$$;

CREATE OR REPLACE FUNCTION app_locatario_id()
RETURNS uuid
LANGUAGE sql STABLE
AS $$
  SELECT NULLIF(current_setting('app.locatario_id', true), '')::uuid
$$;

CREATE OR REPLACE FUNCTION app_is_plataforma()
RETURNS boolean
LANGUAGE sql STABLE
AS $$
  SELECT COALESCE(NULLIF(current_setting('app.is_plataforma', true), ''), 'false')::boolean
$$;

-- ============================================================
-- 2) Definir contexto com regra:
--    - usuário comum: só pode usar o próprio locatário
--    - plataforma: pode usar locatário alvo (ou NULL para modo plataforma)
-- ============================================================

CREATE OR REPLACE FUNCTION app_definir_contexto(
  p_usuario_id uuid,
  p_locatario_id uuid,
  p_is_plataforma boolean
)
RETURNS void
LANGUAGE plpgsql
AS $$
DECLARE
  user_locatario uuid;
BEGIN
  SELECT u.locatario_id INTO user_locatario
  FROM usuarios u
  WHERE u.usuario_id = p_usuario_id
    AND u.excluido_em IS NULL;

  IF p_usuario_id IS NULL THEN
    RAISE EXCEPTION 'p_usuario_id é obrigatório';
  END IF;

  IF user_locatario IS NULL THEN
    -- Usuário de plataforma
    IF NOT p_is_plataforma THEN
      RAISE EXCEPTION 'usuário plataforma deve usar p_is_plataforma=true';
    END IF;

    PERFORM set_config('app.usuario_id', p_usuario_id::text, true);
    PERFORM set_config('app.is_plataforma', 'true', true);

    IF p_locatario_id IS NULL THEN
      PERFORM set_config('app.locatario_id', '', true);
    ELSE
      PERFORM set_config('app.locatario_id', p_locatario_id::text, true);
    END IF;

  ELSE
    -- Usuário comum: só pode atuar no próprio locatário
    IF p_locatario_id IS DISTINCT FROM user_locatario THEN
      RAISE EXCEPTION 'usuário comum só pode atuar no próprio locatário';
    END IF;

    IF p_is_plataforma THEN
      RAISE EXCEPTION 'usuário comum não pode definir is_plataforma=true';
    END IF;

    PERFORM set_config('app.usuario_id', p_usuario_id::text, true);
    PERFORM set_config('app.is_plataforma', 'false', true);
    PERFORM set_config('app.locatario_id', user_locatario::text, true);
  END IF;
END;
$$;

-- ============================================================
-- 3) Helper predicates (para não repetir lógica em policies)
-- ============================================================

-- Em modo "tenant" (contexto setado), isola por locatario_id.
-- Em modo "plataforma" (sem locatario), permite tudo se is_plataforma=true.
CREATE OR REPLACE FUNCTION rls_permite_tabela_tenant(p_row_locatario_id uuid)
RETURNS boolean
LANGUAGE sql STABLE
AS $$
  SELECT
    CASE
      WHEN app_locatario_id() IS NOT NULL THEN (p_row_locatario_id = app_locatario_id())
      ELSE app_is_plataforma() = true
    END
$$;

-- Para CHECK (INSERT/UPDATE) em tabelas tenant, exige que:
-- - Se contexto tenant: row.locatario_id = app_locatario_id()
-- - Se modo plataforma: exige que app_is_plataforma=true E app_locatario_id() NÃO seja NULL
--   (ou seja: plataforma só escreve "dentro de um tenant" quando escolheu locatário)
CREATE OR REPLACE FUNCTION rls_check_tabela_tenant(p_row_locatario_id uuid)
RETURNS boolean
LANGUAGE sql STABLE
AS $$
  SELECT
    CASE
      WHEN app_locatario_id() IS NOT NULL THEN (p_row_locatario_id = app_locatario_id())
      ELSE false
    END
$$;

-- ============================================================
-- 4) RLS por tabela (onde faz sentido)
-- ============================================================

-- ---------- LOCATARIOS (plataforma: tudo; tenant: só o próprio) ----------
ALTER TABLE locatarios ENABLE ROW LEVEL SECURITY;

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'public' AND tablename = 'locatarios' AND policyname = 'p_locatarios_plataforma_all'
  ) THEN
    EXECUTE $sql$
      CREATE POLICY p_locatarios_plataforma_all
      ON locatarios
      TO multilocatario_aplicacao, multilocatario_leitor, multilocatario_auditor, multilocatario_operador_job
      USING (app_is_plataforma() = true AND app_locatario_id() IS NULL)
      WITH CHECK (app_is_plataforma() = true AND app_locatario_id() IS NULL)
    $sql$;
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM pg_policies
    WHERE schemaname = 'public' AND tablename = 'locatarios' AND policyname = 'p_locatarios_tenant_own'
  ) THEN
    EXECUTE $sql$

===== docs/banco-de-dados/postgresql/funcoes/definir_criado_e_atualizado_em.sql =====

-- =========================================================
-- util: timestamps
-- =========================================================
CREATE OR REPLACE FUNCTION fn_definir_criado_e_atualizado_em()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    IF NEW.criado_em IS NULL THEN NEW.criado_em := now(); END IF;
    NEW.atualizado_em := now();
  ELSIF TG_OP = 'UPDATE' THEN
    NEW.atualizado_em := now();
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Como setar no início de cada request (backend)
-- Request de locatário (admin do locatário e usuários do locatário)
-- app.scope = 'TENANT'
-- app.tenant_id = <locatario_id>
-- app.user_id = <usuario_id>
-- opcionais: request_id, ip, user_agent
-- Request de plataforma (superadmin)
-- app.scope = 'PLATAFORMA'
-- app.user_id = <usuario_id_superadmin>
-- não setar app.tenant_id
-- opcionais: request_id, ip, user_agent
-- se a ação afetar um tenant, você passa isso como dado: locatario_alvo_id (coluna do audit)
-- Importante: use SET LOCAL (ou set_config(..., true)) dentro da transação do request, para não “vazar” contexto entre requisições em pool de conexões.

CREATE SCHEMA IF NOT EXISTS app;

CREATE OR REPLACE FUNCTION app.ctx_user_id()
RETURNS uuid
LANGUAGE sql
AS $$
  SELECT nullif(current_setting('app.user_id', true), '')::uuid;
$$;

CREATE OR REPLACE FUNCTION app.ctx_scope()
RETURNS text
LANGUAGE sql
AS $$
  SELECT nullif(current_setting('app.scope', true), '');
$$;

CREATE OR REPLACE FUNCTION app.ctx_tenant_id()
RETURNS uuid
LANGUAGE sql
AS $$
  SELECT nullif(current_setting('app.tenant_id', true), '')::uuid;
$$;

CREATE OR REPLACE FUNCTION app.ctx_request_id()
RETURNS text
LANGUAGE sql
AS $$
  SELECT nullif(current_setting('app.request_id', true), '');
$$;

CREATE OR REPLACE FUNCTION app.ctx_ip()
RETURNS text
LANGUAGE sql
AS $$
  SELECT nullif(current_setting('app.ip', true), '');
$$;

CREATE OR REPLACE FUNCTION app.ctx_user_agent()
RETURNS text
LANGUAGE sql
AS $$
  SELECT nullif(current_setting('app.user_agent', true), '');
$$;

















CREATE OR REPLACE FUNCTION app.assert_scope(expected text)
RETURNS void
LANGUAGE plpgsql
AS $$
BEGIN
  IF app.ctx_scope() IS DISTINCT FROM expected THEN
    RAISE EXCEPTION 'Contexto inválido: esperado scope %, atual %', expected, app.ctx_scope();
  END IF;
END;
$$;

CREATE OR REPLACE FUNCTION app.assert_user()
RETURNS void
LANGUAGE plpgsql
AS $$
BEGIN
  IF app.ctx_user_id() IS NULL THEN
    RAISE EXCEPTION 'Contexto inválido: app.user_id não configurado';
  END IF;
END;
$$;

CREATE OR REPLACE FUNCTION app.assert_tenant()
RETURNS void
LANGUAGE plpgsql
AS $$
BEGIN
  PERFORM app.assert_scope('TENANT');
  IF app.ctx_tenant_id() IS NULL THEN
    RAISE EXCEPTION 'Contexto inválido: app.tenant_id não configurado';
  END IF;
END;
$$;

CREATE OR REPLACE FUNCTION app.assert_platform()
RETURNS void
LANGUAGE plpgsql
AS $$
BEGIN
  PERFORM app.assert_scope('PLATAFORMA');
  -- por segurança, garanta que tenant_id não esteja setado no modo plataforma
  IF app.ctx_tenant_id() IS NOT NULL THEN
    RAISE EXCEPTION 'Contexto inválido: app.tenant_id não deve estar setado em PLATAFORMA';
  END IF;
END;
$$;

  AND locatario_id = app.ctx_tenant_id()

CREATE OR REPLACE FUNCTION fn_validar_locatario_usuario_papel()
RETURNS TRIGGER AS $$
DECLARE
  locatario_usuario UUID;
BEGIN
  SELECT locatario_id INTO locatario_usuario
  FROM usuarios
  WHERE id = NEW.usuario_id;

  IF NEW.locatario_id IS NOT NULL
     AND locatario_usuario IS NOT NULL
     AND NEW.locatario_id <> locatario_usuario THEN
    RAISE EXCEPTION 'Usuário e papel pertencem a locatários diferentes';
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ASSINATURA DE PLANO POR LOCATÁRIO
-- Representa o contrato ativo entre locatário e plano.
--
-- REGRAS DE NEGÓCIO CRÍTICAS:
-- - Um locatário pode ter várias assinaturas históricas
-- - Apenas UMA assinatura pode estar ATIVA por locatário
-- - Mudança de plano cria nova assinatura (não sobrescreve histórico)
-- - Cancelamento/suspensão NÃO remove registros
--
-- STATUS:
-- - ATIVA → plano vigente
-- - SUSPENSA → acesso bloqueado temporariamente
-- - CANCELADA → encerramento definitivo
-- - EXPIRADA → fim automático
--
-- CONSISTÊNCIA:
-- - Nunca apagar registros fisicamente
-- - Datas determinam validade
CREATE TABLE core.assinaturas (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  locatario_id        UUID NOT NULL REFERENCES locatarios(id) ON DELETE CASCADE,
  plano_id            UUID NOT NULL REFERENCES planos(id),

  status              TEXT NOT NULL DEFAULT 'ATIVA' CHECK (status IN ('ATIVA','SUSPENSA','CANCELADA','EXPIRADA')),

  inicio_em           TIMESTAMPTZ NOT NULL DEFAULT now(),
  fim_em              TIMESTAMPTZ NULL,              -- se houver término programado
  renovacao_em        TIMESTAMPTZ NULL,              -- se você quiser “ciclo”
  motivo              TEXT NULL,                     -- por que mudou/cancelou

 -- Cobrança (mínimo)
  moeda                 TEXT NOT NULL DEFAULT 'BRL' CHECK (moeda IN ('BRL','USD','EUR')),
  valor_centavos        INTEGER NOT NULL DEFAULT 0 CHECK (valor_centavos >= 0),
  ciclo                 TEXT NOT NULL DEFAULT 'MENSAL' CHECK (ciclo IN ('MENSAL','ANUAL')),

  proximo_vencimento_em TIMESTAMPTZ NULL,
  ultimo_pagamento_em   TIMESTAMPTZ NULL,

  falha_pagamento_em    TIMESTAMPTZ NULL,

  criado_em           TIMESTAMPTZ NOT NULL DEFAULT now(),
  atualizado_em       TIMESTAMPTZ NOT NULL DEFAULT now(),

  excluido_em         TIMESTAMPTZ NULL               -- exclusão lógica (se quiser)
);

CREATE UNIQUE INDEX ux_assinatura_ativa_por_locatario
ON assinaturas(locatario_id)
WHERE status = 'ATIVA' AND excluido_em IS NULL;

-- ============================================================
-- 5) P3 - COERÊNCIA COM CONTEXTO (app_*)
-- ============================================================

-- Auditoria do locatário só pode ser escrita em modo tenant (não-plataforma) e com locatário definido,
-- e deve bater: locatario_id = app_locatario_id() e usuario_id = app_usuario_id().
CREATE OR REPLACE FUNCTION trg_auditoria_locatario_validar_contexto()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  ctx_locatario uuid;
  ctx_usuario uuid;
BEGIN
  ctx_locatario := app_locatario_id();
  ctx_usuario := app_usuario_id();

  IF app_is_plataforma() THEN
    RAISE EXCEPTION 'auditoria_locatario não aceita modo plataforma; use auditoria_plataforma';
  END IF;

  IF ctx_locatario IS NULL THEN
    RAISE EXCEPTION 'auditoria_locatario exige app_locatario_id() definido';
  END IF;

  IF NEW.locatario_id <> ctx_locatario THEN
    RAISE EXCEPTION 'locatario_id da auditoria (%) difere do contexto (%)', NEW.locatario_id, ctx_locatario;
  END IF;

  IF ctx_usuario IS NOT NULL AND NEW.usuario_id <> ctx_usuario THEN
    RAISE EXCEPTION 'usuario_id da auditoria (%) difere do contexto (%)', NEW.usuario_id, ctx_usuario;
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_auditoria_locatario_contexto') THEN
    CREATE TRIGGER tg_auditoria_locatario_contexto
    BEFORE INSERT ON auditoria_locatario
    FOR EACH ROW
    EXECUTE FUNCTION trg_auditoria_locatario_validar_contexto();
  END IF;
END;
$$;

  RETURN NEW;
END;
$$;

-- ============================================================
-- 5) P3 - COERÊNCIA COM CONTEXTO (app_*)
-- ============================================================

-- Auditoria de plataforma só pode ser escrita em modo plataforma.
-- Se locatario_alvo_id vier NULL, preenche a partir do app_locatario_id() (pode continuar NULL).
CREATE OR REPLACE FUNCTION trg_auditoria_plataforma_validar_contexto()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  ctx_locatario uuid;
  ctx_usuario uuid;
BEGIN
  ctx_locatario := app_locatario_id();
  ctx_usuario := app_usuario_id();

  IF NOT app_is_plataforma() THEN
    RAISE EXCEPTION 'auditoria_plataforma exige app_is_plataforma()=true';
  END IF;

  IF ctx_usuario IS NOT NULL AND NEW.usuario_id <> ctx_usuario THEN
    RAISE EXCEPTION 'usuario_id da auditoria (%) difere do contexto (%)', NEW.usuario_id, ctx_usuario;
  END IF;

  IF NEW.locatario_alvo_id IS NULL THEN
    NEW.locatario_alvo_id := ctx_locatario; -- pode permanecer NULL
  ELSE
    -- Se o contexto tiver locatário definido, ele deve coincidir com o alvo (evita inconsistência)
    IF ctx_locatario IS NOT NULL AND NEW.locatario_alvo_id <> ctx_locatario THEN
      RAISE EXCEPTION 'locatario_alvo_id (%) difere do contexto (%)', NEW.locatario_alvo_id, ctx_locatario;
    END IF;
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_auditoria_plataforma_contexto') THEN
    CREATE TRIGGER tg_auditoria_plataforma_contexto
    BEFORE INSERT ON auditoria_plataforma
    FOR EACH ROW
    EXECUTE FUNCTION trg_auditoria_plataforma_validar_contexto();
  END IF;
END;
$$;

RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  -- UPDATE nunca é permitido
  IF TG_OP = 'UPDATE' THEN
    RAISE EXCEPTION 'auditoria é append-only: UPDATE não permitido (%)', TG_TABLE_NAME;
  END IF;

  -- DELETE permitido somente para o purgador
  IF TG_OP = 'DELETE' THEN
    IF current_user = 'multilocatario_purgador' THEN
      RETURN OLD;
    END IF;
    RAISE EXCEPTION 'auditoria é append-only: DELETE permitido apenas para multilocatario_purgador (%)', TG_TABLE_NAME;
  END IF;

  RETURN NEW;
END;
$$;

    CREATE TRIGGER tg_auditoria_plataforma_immut

  telefone TEXT NULL,

  atualizado_em TIMESTAMPTZ NOT NULL DEFAULT now(),

    FOREIGN KEY (locatario_id)

    ON UPDATE CASCADE
    ON DELETE CASCADE;
  END IF;

$$;

-- CONFIGURAÇÕES DO LOCATÁRIO (ENTIDADE FRACA)
-- Representa preferências e identidade visual do locatário.
--
-- REGRAS:
-- - NÃO existe sem locatário
-- - Pode haver múltiplas configurações (histórico / perfis)
-- - Apenas uma pode estar ativa (regra de negócio, não estrutural)
-- - Alterações NÃO impactam plano ou assinatura
--
-- EXEMPLOS:
-- - identidade visual
-- - emails
-- - tema
CREATE TABLE IF NOT EXISTS core.configuracoes_locatario (
  locatario_id       UUID NOT NULL,
  configuracao_id    UUID NOT NULL,
  modo_tema          TEXT NOT NULL DEFAULT 'SISTEMA' CHECK (modo_tema IN ('CLARO','ESCURO','SISTEMA')),
  identidade_visual  TEXT UNIQUE NOT NULL CHECK(length(trim(identidade_visual))>0), -- nome da identidade (padrão, natal, ano novo, etc)
  cor_primaria       TEXT NULL, -- ex: "#3B82F6"
  cor_secundaria     TEXT NULL, -- ex: "#3B82F6"
  cor_terciaria      TEXT NULL, -- ex: "#3B82F6"
  url_logo           TEXT NULL, -- logomarca da empresa
  ativa              BOOLEAN NOT NULL DEFAULT FALSE, -- somente 1 configuração ativa por locatário;
  criado_em          TIMESTAMPTZ NOT NULL DEFAULT now(),
  atualizado_em      TIMESTAMPTZ NOT NULL DEFAULT now()  -- atualizado por trigger
);

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'pk_configuracoes_locatario') THEN

-- ============================================================
-- P3: Sempre 1 configuração ativa por locatário
-- - Inseriu a primeira config do locatário => força ativa=true
-- - Ao ativar uma config => desativa as demais do mesmo locatário
-- - Não permite ficar com 0 ativas (bloqueia desativação da última ativa)
-- ============================================================
CREATE OR REPLACE FUNCTION trg_configuracoes_locatario_p3_ativa_unica()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  has_any boolean;
  has_other_active boolean;
BEGIN
  -- primeira configuração do locatário deve ser ativa
  IF TG_OP = 'INSERT' THEN
    SELECT EXISTS (
      SELECT 1 FROM configuracoes_locatario
      WHERE locatario_id = NEW.locatario_id
    ) INTO has_any;

    IF NOT has_any THEN
      NEW.ativa := TRUE;
    END IF;

    IF NEW.ativa THEN
      UPDATE configuracoes_locatario
      SET ativa = FALSE
      WHERE locatario_id = NEW.locatario_id
        AND configuracao_id IS DISTINCT FROM NEW.configuracao_id;
    END IF;

    RETURN NEW;
  END IF;

  -- update: ativar => desativar as demais; desativar última ativa => bloquear
  IF TG_OP = 'UPDATE' THEN
    IF NEW.ativa AND NOT OLD.ativa THEN
      UPDATE configuracoes_locatario
      SET ativa = FALSE
      WHERE locatario_id = NEW.locatario_id
        AND configuracao_id <> NEW.configuracao_id;
      RETURN NEW;
    END IF;

    IF (NOT NEW.ativa) AND OLD.ativa THEN
      SELECT EXISTS (
        SELECT 1
        FROM configuracoes_locatario
        WHERE locatario_id = NEW.locatario_id
          AND ativa = TRUE
          AND configuracao_id <> NEW.configuracao_id
      ) INTO has_other_active;

      IF NOT has_other_active THEN
        RAISE EXCEPTION 'não é permitido deixar o locatário sem configuração ativa';
      END IF;

      RETURN NEW;
    END IF;

    RETURN NEW;
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_configuracoes_locatario_p3') THEN
    CREATE TRIGGER tg_configuracoes_locatario_p3
    BEFORE INSERT OR UPDATE ON configuracoes_locatario
    FOR EACH ROW
    EXECUTE FUNCTION trg_configuracoes_locatario_p3_ativa_unica();
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_configuracoes_locatario_set_atualizado_em') THEN
    CREATE TRIGGER tg_configuracoes_locatario_set_atualizado_em
    BEFORE INSERT OR UPDATE ON configuracoes_locatario
    FOR EACH ROW
    EXECUTE FUNCTION trg_set_atualizado_em();
  END IF;
END;
$$;

-- Representa uma empresa/cliente isolado logicamente na plataforma.

-- - Suspensão altera apenas o status, nunca remove dados
-- - Exclusão é sempre lógica (soft delete)

-- IDENTIDADE:
-- - slug é único globalmente
-- - slug é usado em URLs/subdomínios
-- - slug NÃO deve ser alterado após criação
--
-- CONSISTÊNCIA:
-- - Se excluido_em estiver preenchido, o locatário é considerado inativo
-- - status controla acesso ao sistema (ATIVO/SUSPENSO/etc)
--
-- NÃO CONFUNDIR COM:
-- - Usuários
-- - Assinaturas
-- - Planos

-- Máscara/validação do slug (regex) + tamanho

-- ============================================================
-- P2: Consistência de ciclo de vida + bloqueio de alteração do slug
-- - slug sempre normalizado (lower + trim)
-- - slug não pode ser alterado após criação
-- - Se suspenso_em estiver preenchido, status deve ser "SUSPENSO" (se existir no enum),
--   senão pelo menos não pode ser "ATIVO"
-- - Se excluido_em estiver preenchido, status deve ser "EXCLUIDO" (se existir no enum),
--   senão pelo menos não pode ser "ATIVO"
-- - Reativação não pode ocorrer se excluído (excluido_em não pode estar preenchido)
-- - Se status=ATIVO, suspenso_em deve ser NULL
-- ============================================================
CREATE OR REPLACE FUNCTION trg_locatarios_p2_consistencia()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  has_suspenso boolean;
  has_excluido boolean;
BEGIN
  -- normalização de slug
  NEW.slug := lower(trim(NEW.slug));

  -- máscara (defensiva; o CHECK também cobre)
  IF NEW.slug IS NULL OR NEW.slug = '' THEN
    RAISE EXCEPTION 'slug é obrigatório';
  END IF;

  IF length(NEW.slug) < 3 OR length(NEW.slug) > 63 OR NEW.slug !~ '^[a-z0-9]+(?:-[a-z0-9]+)*$' THEN
    RAISE EXCEPTION 'slug inválido: use 3..63 chars, [a-z0-9] e hífen, sem espaços (ex: minha-loja-01)';
  END IF;

  -- slug imutável após criação
  IF TG_OP = 'UPDATE' AND NEW.slug IS DISTINCT FROM OLD.slug THEN
    RAISE EXCEPTION 'slug não pode ser alterado após criação';
  END IF;

  -- enum labels opcionais (evita quebrar caso o enum não tenha as labels)
  SELECT EXISTS (
    SELECT 1
    FROM pg_enum e
    JOIN pg_type t ON t.oid = e.enumtypid
    WHERE t.typname = 'status_locatario' AND e.enumlabel = 'SUSPENSO'
  ) INTO has_suspenso;

  SELECT EXISTS (
    SELECT 1
    FROM pg_enum e
    JOIN pg_type t ON t.oid = e.enumtypid
    WHERE t.typname = 'status_locatario' AND e.enumlabel = 'EXCLUIDO'
  ) INTO has_excluido;

  -- reativação: não pode se estiver excluído
  IF NEW.reativado_em IS NOT NULL AND NEW.excluido_em IS NOT NULL THEN
    RAISE EXCEPTION 'não é permitido reativar um locatário excluído (excluido_em preenchido)';
  END IF;

  -- se reativado_em foi preenchido: força ATIVO e limpa suspensão
  IF NEW.reativado_em IS NOT NULL THEN
    NEW.status := 'ATIVO';
    NEW.suspenso_em := NULL;
    NEW.motivo_suspensao := NULL;
  END IF;

  -- exclusão lógica implica "não ativo"
  IF NEW.excluido_em IS NOT NULL THEN
    IF has_excluido THEN
      NEW.status := 'EXCLUIDO';
    ELSE
      IF NEW.status = 'ATIVO' THEN
        RAISE EXCEPTION 'locatário excluído não pode permanecer com status ATIVO';
      END IF;
    END IF;
  END IF;

  -- suspensão implica "não ativo"
  IF NEW.suspenso_em IS NOT NULL AND NEW.excluido_em IS NULL THEN
    IF has_suspenso THEN
      NEW.status := 'SUSPENSO';
    ELSE
      IF NEW.status = 'ATIVO' THEN
        RAISE EXCEPTION 'locatário suspenso não pode permanecer com status ATIVO';
      END IF;
    END IF;
  END IF;

  -- status ATIVO implica não suspenso
  IF NEW.status = 'ATIVO' THEN
    NEW.suspenso_em := NULL;
    NEW.motivo_suspensao := NULL;
  END IF;

  -- se status SUSPENSO e suspenso_em não veio preenchido, preenche automaticamente (se label existir)
  IF has_suspenso AND NEW.status = 'SUSPENSO' AND NEW.suspenso_em IS NULL THEN
    NEW.suspenso_em := now();
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_locatarios_p2') THEN
    CREATE TRIGGER tg_locatarios_p2
    BEFORE INSERT OR UPDATE ON locatarios
    FOR EACH ROW
    EXECUTE FUNCTION trg_locatarios_p2_consistencia();
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_locatarios_set_atualizado_em') THEN
    CREATE TRIGGER tg_locatarios_set_atualizado_em
    BEFORE INSERT OR UPDATE ON locatarios
    FOR EACH ROW
    EXECUTE FUNCTION trg_set_atualizado_em();
  END IF;
END;
$$;

-- ============================================================
-- Proteção de papéis canônicos (não renomear / não deletar)
-- ============================================================

CREATE OR REPLACE FUNCTION trg_papeis_proteger_canonicos()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  -- Bloqueia DELETE
  IF TG_OP = 'DELETE' THEN
    IF OLD.nome IN ('SUPERADMIN','ADMIN','GERENTE','OPERADOR') THEN
      RAISE EXCEPTION 'papel canônico não pode ser removido: %', OLD.nome;
    END IF;
    RETURN OLD;
  END IF;

  -- Bloqueia renomear (UPDATE nome)
  IF TG_OP = 'UPDATE' THEN
    IF OLD.nome IN ('SUPERADMIN','ADMIN','GERENTE','OPERADOR')
       AND NEW.nome IS DISTINCT FROM OLD.nome THEN
      RAISE EXCEPTION 'papel canônico não pode ser renomeado: %', OLD.nome;
    END IF;
    RETURN NEW;
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_papeis_proteger_canonicos') THEN
    CREATE TRIGGER tg_papeis_proteger_canonicos
    BEFORE UPDATE OR DELETE ON papeis
    FOR EACH ROW
    EXECUTE FUNCTION trg_papeis_proteger_canonicos();
  END IF;
END;
$$;

-- =========================================================
-- TOKENS DE AUTENTICAÇÃO / SESSÃO
--
-- Representa tokens emitidos pelo sistema de autenticação.
--
-- REGRAS CRÍTICAS DE SEGURANÇA:
-- - Tokens podem ser revogados
-- - Tokens possuem validade temporal
-- - NÃO confiar apenas em expiração do JWT
-- - Nunca armazenar tokens sensíveis em texto puro
-- - NÃO Armazenar JWT puro
-- - NÃO Confiar apenas em expiração
-- - NÃO Permitir múltiplos refresh tokens ativos sem controle
--
-- USO:
-- - Controle de sessão
-- - Rotação / revogação
-- =========================================================
CREATE TABLE IF NOT EXISTS core.tokens_locatario (
  token_id           UUID NOT NULL DEFAULT gen_random_uuid(),
  usuario_id         UUID NOT NULL,
  locatario_id       UUID NOT NULL, -- NULL indica usuário de nível plataforma (ex.: superadmin)
  hash_token         TEXT UNIQUE NOT NULL, -- nunca guarde o token em claro
  tipo               tipo_token NOT NULL, -- access / refresh / etc.
  emitido_em         TIMESTAMPTZ NOT NULL DEFAULT now(),
  expira_em          TIMESTAMPTZ NOT NULL,
  revogado_em        TIMESTAMPTZ NULL,
  proximo_token_id   UUID NULL, -- id do novo token (rotação)
  proximo_usuario_id UUID NULL,
  proximo_locatario_id UUID NULL,
  agente_usuario     TEXT NULL,
  endereco_ip        INET NULL
);

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'pk_token') THEN
    ALTER TABLE tokens ADD CONSTRAINT pk_token PRIMARY KEY (token_id, usuario_id, locatario_id);
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'fk_proximo_token_id') THEN
    ALTER TABLE tokens
    ADD CONSTRAINT fk_proximo_token_id
    FOREIGN KEY (proximo_token_id, proximo_usuario_id, proximo_locatario_id)
    REFERENCES tokens(token_id, usuario_id, locatario_id)
    ON UPDATE CASCADE
    ON DELETE SET NULL;
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'fk_tokens_usuarios') THEN
    ALTER TABLE tokens
    ADD CONSTRAINT fk_tokens_usuarios
    FOREIGN KEY (usuario_id)
    REFERENCES usuarios(usuario_id)
    ON UPDATE CASCADE
    ON DELETE CASCADE;
  END IF;

  IF NOT EXISTS (SELECT 1 FROM pg_constraint WHERE conname = 'fk_tokens_atualizacao_locatarios') THEN
    ALTER TABLE tokens
    ADD CONSTRAINT fk_tokens_atualizacao_locatarios
    FOREIGN KEY (locatario_id)
    REFERENCES locatarios(locatario_id)
    ON UPDATE CASCADE
    ON DELETE CASCADE;
  END IF;
END;
$$;

-- ============================================================
-- DELTA TOKENS COERÊNCIA USUÁRIO x LOCATÁRIO (v6)
-- Objetivo: impedir tokens com locatario_id diferente do locatario_id do usuário
-- Regra:
-- - Usuário comum (usuarios.locatario_id IS NOT NULL): tokens.locatario_id deve ser igual
-- - Usuário de plataforma (usuarios.locatario_id IS NULL): permitido tokens com locatario_id (quando atuando em tenant)
--   e também permitido NULL (se você emitir token puramente de plataforma; opcional)
-- ============================================================

CREATE OR REPLACE FUNCTION trg_tokens_validar_locatario_do_usuario()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  user_locatario uuid;
BEGIN
  SELECT u.locatario_id INTO user_locatario
  FROM usuarios u
  WHERE u.usuario_id = NEW.usuario_id
    AND u.excluido_em IS NULL;

  IF user_locatario IS NULL THEN
    -- Usuário de plataforma (SUPERADMIN): pode emitir token para um tenant (locatario_id preenchido),
    -- ou eventualmente um token de plataforma (locatario_id NULL), se você permitir.
    RETURN NEW;
  END IF;

  -- Usuário comum: token deve ser do mesmo locatário do usuário
  IF NEW.locatario_id IS NULL THEN
    RAISE EXCEPTION 'token inválido: usuário comum exige locatario_id (tenant)';
  END IF;

  IF NEW.locatario_id <> user_locatario THEN
    RAISE EXCEPTION 'token inválido: locatario_id do token (%) difere do locatario_id do usuário (%)',
      NEW.locatario_id, user_locatario;
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_tokens_validar_locatario_do_usuario') THEN
    CREATE TRIGGER tg_tokens_validar_locatario_do_usuario
    BEFORE INSERT OR UPDATE ON tokens
    FOR EACH ROW
    EXECUTE FUNCTION trg_tokens_validar_locatario_do_usuario();
  END IF;
END;
$$;

-- ============================================================
-- 3.1) no máximo 1 ADMIN por locatário (trigger em usuarios_papeis)
-- ============================================================

CREATE OR REPLACE FUNCTION trg_usuarios_papeis_max_um_admin_por_locatario()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  admin_papel_id uuid;
  admin_conflict boolean;
BEGIN
  -- Descobre o ID do papel ADMIN (canônico por nome)
  SELECT id INTO admin_papel_id
  FROM papeis
  WHERE nome = 'ADMIN'
  LIMIT 1;

  IF admin_papel_id IS NULL THEN
    RAISE EXCEPTION 'papel canônico ADMIN não existe em papeis; crie-o antes de aplicar a regra';
  END IF;

  -- Só valida se a operação envolve ADMIN
  IF NEW.papel_id <> admin_papel_id THEN
    RETURN NEW;
  END IF;

  -- Regra: no máximo 1 usuário diferente com ADMIN no mesmo locatário
  SELECT EXISTS (
    SELECT 1
    FROM usuarios_papeis up
    WHERE up.locatario_id = NEW.locatario_id
      AND up.papel_id = admin_papel_id
      AND up.usuario_id <> NEW.usuario_id
  ) INTO admin_conflict;

  IF admin_conflict THEN
    RAISE EXCEPTION 'regra violada: já existe um ADMIN para este locatário (no máximo 1)';
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_usuarios_papeis_max_um_admin') THEN
    CREATE TRIGGER tg_usuarios_papeis_max_um_admin
    BEFORE INSERT OR UPDATE ON usuarios_papeis
    FOR EACH ROW
    EXECUTE FUNCTION trg_usuarios_papeis_max_um_admin_por_locatario();
  END IF;
END;
$$;

-- ============================================================
-- 2) usuarios_plataforma singleton + consistência com usuarios.locatario_id IS NULL
-- ============================================================

CREATE OR REPLACE FUNCTION trg_usuarios_plataforma_singleton_consistencia()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  already_exists boolean;
  is_platform_user_ok boolean;
BEGIN
  -- Singleton: no máximo 1 linha
  SELECT EXISTS (
    SELECT 1 FROM usuarios_plataforma
    WHERE usuario_id <> NEW.usuario_id
  ) INTO already_exists;

  IF already_exists THEN
    RAISE EXCEPTION 'usuarios_plataforma é singleton: já existe um SUPERADMIN cadastrado';
  END IF;

  -- Consistência: usuário deve ter locatario_id IS NULL
  SELECT EXISTS (
    SELECT 1
    FROM usuarios u
    WHERE u.usuario_id = NEW.usuario_id
      AND u.locatario_id IS NULL
      AND u.excluido_em IS NULL
  ) INTO is_platform_user_ok;

  IF NOT is_platform_user_ok THEN
    RAISE EXCEPTION 'usuário de plataforma inválido: o registro em usuarios deve existir, não estar excluído e locatario_id deve ser NULL';
  END IF;

  RETURN NEW;
END;
$$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = 'tg_usuarios_plataforma_singleton') THEN
    CREATE TRIGGER tg_usuarios_plataforma_singleton
    BEFORE INSERT OR UPDATE ON usuarios_plataforma
    FOR EACH ROW
    EXECUTE FUNCTION trg_usuarios_plataforma_singleton_consistencia();
  END IF;
END;
$$;

-- Tipos (ENUM) - opcional/recomendado
DO $$
BEGIN
-- =========================================================
-- ENUM: status_locatario
-- Representa o estado operacional do locatário (tenant).
--
-- REGRAS DE NEGÓCIO:
-- - Controla acesso do locatário ao sistema
-- - NÃO indica remoção física de dados
-- - Exclusão é sempre lógica
--
-- ESTADOS:
-- - ATIVO     → locatário operacional e com acesso permitido
-- - SUSPENSO  → acesso temporariamente bloqueado (inadimplência, suporte, etc)
-- - EXCLUIDO  → exclusão lógica / desativação definitiva
--
-- CONSISTÊNCIA:
-- - EXCLUIDO normalmente implica excluido_em preenchido
-- - Registros nunca devem ser removidos fisicamente
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'status_locatario') THEN
    CREATE TYPE status_locatario AS ENUM ('ATIVO','SUSPENSO','EXCLUIDO');
  END IF;

-- =========================================================
-- ENUM: planos_locatario
-- Classificação comercial de planos disponíveis.
--
-- REGRAS:
-- - Valores representam categorias de produto, não regras técnicas
-- - Mudanças de plano NÃO alteram histórico
--
-- EXEMPLOS:
-- - GRATUITO    → acesso básico / onboarding
-- - BASICO      → plano de entrada
-- - PRO         → funcionalidades avançadas
-- - EMPRESARIAL → uso intensivo / limites maiores
--
-- IMPORTANTE:
-- - Limites reais devem ser controlados por tabela de limites
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'planos_locatario') THEN
    CREATE TYPE planos_locatario AS ENUM ('GRATUITO','BASICO','PRO','EMPRESARIAL');
  END IF; 

-- =========================================================
-- ENUM: status_usuario
-- Estado operacional da conta do usuário.
--
-- REGRAS:
-- - NÃO é exclusão lógica
-- - Controla autenticação/autorização
--
-- ESTADOS:
-- - ATIVO      → usuário autenticável
-- - BLOQUEADO  → acesso negado sem remover vínculos
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'status_usuario') THEN
    CREATE TYPE status_usuario AS ENUM ('ATIVO','BLOQUEADO');
  END IF;

-- =========================================================
-- ENUM: tipo_origem_importacao
-- Origem dos dados em rotinas de importação.
--
-- USO:
-- - Controla pipeline de parsing/validação
-- - NÃO representa formato de armazenamento interno
--
-- VALORES:
-- - CSV
-- - XLSX
-- - GOOGLE_SHEETS
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'tipo_origem_importacao') THEN
    CREATE TYPE tipo_origem_importacao AS ENUM ('CSV', 'XLSX', 'GOOGLE_SHEETS');
  END IF;

-- =========================================================
-- ENUM: status_job
-- Máquina de estados de jobs assíncronos.
--
-- REGRAS IMPORTANTES:
-- - Jobs NÃO devem pular estados arbitrariamente
-- - SUCESSO e ERRO são estados terminais
--
-- TRANSIÇÕES ESPERADAS:
-- PENDENTE → EXECUTANDO
-- EXECUTANDO → SUCESSO | ERRO | CANCELADO
--
-- ESTADOS:
-- - PENDENTE     → aguardando execução
-- - EXECUTANDO   → em processamento
-- - SUCESSO      → concluído com êxito (terminal)
-- - ERRO         → falha definitiva (terminal)
-- - CANCELADO    → encerrado manualmente
-- =========================================================

  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'status_job') THEN
    CREATE TYPE status_job AS ENUM ('PENDENTE', 'EXECUTANDO', 'SUCESSO', 'ERRO', 'CANCELADO');
  END IF;

-- =========================================================
-- ENUM: formato_exportacao
-- Formato de saída em rotinas de exportação.
--
-- REGRAS:
-- - Controla serialização/renderização
-- - NÃO define tipo MIME automaticamente
--
-- VALORES:
-- - CSV
-- - XLSX
-- - PDF
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'formato_exportacao') THEN
    CREATE TYPE formato_exportacao AS ENUM ('CSV', 'XLSX', 'PDF');
  END IF;

-- =========================================================
-- ENUM: status_assinatura
-- Estados da assinatura de um locatário
--
-- REGRAS IMPORTANTES:
-- - Reativação de locatário geralmente é possível
--
-- TRANSIÇÕES ESPERADAS:
-- ATIVA → SUSPENSA | CANCELADA | EXPIRADA
-- SUSPENSA → ATIVA
-- CANCELADA → ATIVA
-- EXPIRADA → ATIVA
--
-- ESTADOS:
-- - ATIVA     → em uso
-- - SUSPENSA  → por decisão do administrador do sistema
-- - CANCELADA → cancelada pelo locatário
-- - EXPIRADA  → chegou ao fim do contrato
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'status_assinatura') THEN
    CREATE TYPE status_assinatura AS ENUM ('ATIVA','SUSPENSA','CANCELADA','EXPIRADA');
  END IF;

-- =========================================================
-- ENUMs recomendados (mínimos para o pacote)
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'ciclo_cobranca') THEN
    CREATE TYPE ciclo_cobranca AS ENUM ('MENSAL','ANUAL');
  END IF;

-- =========================================================
-- ENUMs recomendados (mínimos para o pacote)
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'status_modulo_assinatura') THEN
    CREATE TYPE status_modulo_assinatura AS ENUM ('ATIVO','SUSPENSO','REMOVIDO');
  END IF;

-- =========================================================
-- ENUMs recomendados (mínimos para o pacote)
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'origem_modulo_assinatura') THEN
    CREATE TYPE origem_modulo_assinatura AS ENUM ('PLANO_SUGERIDO','ADDON','CORTESIA','MANUAL_SUPERADMIN');
  END IF;

-- =========================================================
-- ENUMs recomendados (mínimos para o pacote)
-- =========================================================
  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'tipo_token') THEN
    CREATE TYPE tipo_token AS ENUM ('ACESSO','REFRESH','RECUPERACAO_SENHA','CONFIRMACAO_EMAIL');
  END IF;
END $$;

CREATE OR REPLACE FUNCTION fn_locatarios_after_insert()
RETURNS TRIGGER AS $$
BEGIN
  PERFORM fn_auditar_evento(
    'LOCATARIO_CRIADO',
    'CRIAR',
    'locatarios',
    NEW.id::text,
    'SUCESSO',
    jsonb_build_object('nome', NEW.nome)
  );
  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_locatarios_after_update()
RETURNS TRIGGER AS $$
DECLARE
  v_alterados jsonb := '{}'::jsonb;
BEGIN
  -- registra apenas “quais campos mudaram” (seguro e útil)
  IF NEW.nome IS DISTINCT FROM OLD.nome THEN
    v_alterados := v_alterados || jsonb_build_object('nome', jsonb_build_object('de', OLD.nome, 'para', NEW.nome));
  END IF;

  IF NEW.status IS DISTINCT FROM OLD.status THEN
    v_alterados := v_alterados || jsonb_build_object('status', jsonb_build_object('de', OLD.status, 'para', NEW.status));
  END IF;

  PERFORM fn_auditar_evento(
    'LOCATARIO_ATUALIZADO',
    'ATUALIZAR',
    'locatarios',
    NEW.id::text,
    'SUCESSO',
    jsonb_build_object('alteracoes', v_alterados)
  );

  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_revogar_tokens_usuario(p_usuario_id uuid, p_locatario_id uuid, p_motivo text)
RETURNS void AS $$
BEGIN
  UPDATE tokens_atualizacao
     SET revogado_em = now()
   WHERE usuario_id = p_usuario_id
     AND (locatario_id IS NOT DISTINCT FROM p_locatario_id)
     AND revogado_em IS NULL;

  PERFORM fn_auditar_evento(
    'TOKENS_REVOGADOS',
    'ATUALIZAR',
    'tokens_atualizacao',
    NULL,
    'SUCESSO',
    jsonb_build_object('usuario_id', p_usuario_id, 'locatario_id', p_locatario_id, 'motivo', p_motivo)
  );
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_usuarios_after_update()
RETURNS TRIGGER AS $$
DECLARE
  v_alterados jsonb := '{}'::jsonb;
  v_revogar boolean := false;
  v_motivo text := NULL;
BEGIN
  IF NEW.email IS DISTINCT FROM OLD.email THEN
    v_alterados := v_alterados || jsonb_build_object('email', jsonb_build_object('de', OLD.email, 'para', NEW.email));
  END IF;

  IF NEW.status IS DISTINCT FROM OLD.status THEN
    v_alterados := v_alterados || jsonb_build_object('status', jsonb_build_object('de', OLD.status, 'para', NEW.status));
    v_revogar := true;
    v_motivo := 'mudanca_status';
  END IF;

  -- se seu schema tiver senha_hash (ou campo equivalente)
  IF (NEW.senha_hash IS DISTINCT FROM OLD.senha_hash) THEN
    v_alterados := v_alterados || jsonb_build_object('senha', 'alterada');
    v_revogar := true;
    v_motivo := COALESCE(v_motivo, 'mudanca_senha');
  END IF;

  PERFORM fn_auditar_evento(
    'USUARIO_ATUALIZADO',
    'ATUALIZAR',
    'usuarios',
    NEW.id::text,
    'SUCESSO',
    jsonb_build_object('alteracoes', v_alterados)
  );

  IF v_revogar THEN
    PERFORM fn_revogar_tokens_usuario(NEW.id, NEW.locatario_id, v_motivo);
  END IF;

  RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE VIEW vw_modulos_habilitados_locatario AS
SELECT
  l.locatario_id,
  m.codigo,
  'SISTEMA'::TEXT AS origem
FROM locatarios l
JOIN modulos m
  ON m.gratuito = TRUE
 AND m.ativo = TRUE

UNION

SELECT
  ml.locatario_id,
  m.codigo,
  ml.origem
FROM modulos_locatario ml
JOIN modulos m
  ON m.id = ml.modulo_id
WHERE ml.habilitado = TRUE
  AND m.ativo = TRUE;
