function criarDepsAssinaturaModulos() {
  return {
    assinaturasRepositorio: createAssinaturasRepositorioPrisma(clientePrisma),
    assinaturasModulosRepositorio: createAssinaturasModulosRepositorioPrisma(clientePrisma),

function responderErroValidacao(
  reply: FastifyReply,
  parseResult: {
    error: {
      flatten: () => {
        fieldErrors: Record<string, string[] | undefined>;
        formErrors: string[];
      };
    };
  },
) {
  const { fieldErrors, formErrors } = parseResult.error.flatten();

  return sendHttpError(reply, {
    codigo: "INVALIDO",
    mensagem: formErrors[0] ?? "Dados inválidos na requisição.",
    detalhes: {
      operacao: "http.assinaturaModulos.validacao",
      campos: fieldErrors,
      formulario: formErrors,
    },
  });
}

/**
 * 
 * @param request 
 * @param reply 
 * @returns 
 */
export async function listarMeusModulosAssinaturaController(request: FastifyRequest, reply: FastifyReply) {
  const ctx = await autorizarHttp(request, reply, {
    permissao: Permissoes.TENANT_ASSINATURA_LER,
    modulo: "ASSINATURAS",
  });
  if (!ctx) return;

  const { modulosHabilitadosRepositorio } = criarDepsAssinaturaModulos();

  const listarModulosHabilitados = createFnListarModulosHabilitados({
    repositorio: modulosHabilitadosRepositorio,
  });

  const resultado = await listarModulosHabilitados({
    locatarioId: ctx.locatarioId,
  });

  if (!resultado.ok) return sendHttpError(reply, resultado.error);

  return reply.status(200).send({
    locatarioId: resultado.value.locatarioId,
    modulos: resultado.value.modulos.map(toModuloHabilitadoDto),
  });
}

/**
 * 
 * @param request 
 * @param reply 
 * @returns 
 */
export async function vincularModuloNaMinhaAssinaturaController(request: FastifyRequest<{ Body: VincularModuloAssinaturaBodyDto }>, reply: FastifyReply) {
  const ctx = await autorizarHttp(request, reply, {
    permissao: Permissoes.TENANT_ASSINATURA_ATUALIZAR,
    modulo: "ASSINATURAS",
  });
  if (!ctx) return;

  const bodyParse = parseVincularModuloAssinaturaBody(request.body);
  if (!bodyParse.success) return responderErroValidacao(reply, bodyParse);

  const deps = criarDepsAssinaturaModulos();
  const assinaturaAtiva = await deps.assinaturasRepositorio.buscarAssinaturaAtivaPorLocatario(ctx.locatarioId);

  if (!assinaturaAtiva.ok) return sendHttpError(reply, assinaturaAtiva.error);

  if (!assinaturaAtiva.value) {
    return sendHttpError(reply, {
      codigo: "ASSINATURA_NAO_ENCONTRADA",
      mensagem: "Nenhuma assinatura ativa encontrada para este locatário.",
      detalhes: {
        locatarioId: ctx.locatarioId,
      },
    });
  }

  const vincularModuloNaAssinatura = createFnVincularModuloNaAssinatura({
    assinaturasRepositorio: deps.assinaturasRepositorio,
    assinaturasModulosRepositorio: deps.assinaturasModulosRepositorio,
    modulosConsultaRepositorio: deps.modulosConsultaRepositorio,
    relogio: deps.relogio,
    transacao: deps.transacao,
  });

  const resultado = await vincularModuloNaAssinatura({
    assinaturaId: assinaturaAtiva.value.assinaturaId,
    moduloId: bodyParse.data.moduloId,
  });

  await auditarController(request, ctx, {
    evento: "assinatura.modulo.vinculado",
    sucesso: resultado.ok,
    entidadeTipo: "assinatura_modulo",
    entidadeId: bodyParse.data.moduloId,
    erroCodigo: !resultado.ok ? resultado.error.codigo : null,
    erroMensagem: !resultado.ok ? resultado.error.mensagem : null,
    metadados: metaAcao({
      assinaturaId: assinaturaAtiva.value.assinaturaId,
      moduloId: bodyParse.data.moduloId,
    }),
  });

  if (!resultado.ok) return sendHttpError(reply, resultado.error);

  return reply.status(201).send({
    item: toVinculoModuloDto({
      assinaturaId: resultado.value.assinaturaId,
      moduloId: resultado.value.moduloId,
      origem: resultado.value.origem,
      gratuito: resultado.value.gratuito,
      habilitadoEm: resultado.value.habilitadoEm,
      desabilitadoEm: null,
    }),
  });
}

export async function desvincularModuloDaMinhaAssinaturaController(request: FastifyRequest<{ Params: ModuloIdParamsDto }>, reply: FastifyReply) {
  const ctx = await autorizarHttp(request, reply, {
    permissao: Permissoes.TENANT_ASSINATURA_ATUALIZAR,
    modulo: "ASSINATURAS",
  });
  if (!ctx) return;

  const paramsParse = parseModuloIdParams(request.params);
  if (!paramsParse.success) return responderErroValidacao(reply, paramsParse);

  const deps = criarDepsAssinaturaModulos();
  const assinaturaAtiva = await deps.assinaturasRepositorio.buscarAssinaturaAtivaPorLocatario(ctx.locatarioId);

  if (!assinaturaAtiva.ok) return sendHttpError(reply, assinaturaAtiva.error);

  if (!assinaturaAtiva.value) {
    return sendHttpError(reply, {
      codigo: "ASSINATURA_NAO_ENCONTRADA",
      mensagem: "Nenhuma assinatura ativa encontrada para este locatário.",
      detalhes: {
        locatarioId: ctx.locatarioId,
      },
    });
  }

  const desvincularModuloDaAssinatura = createFnDesvincularModuloDaAssinatura({
    assinaturasRepositorio: deps.assinaturasRepositorio,
    assinaturasModulosRepositorio: deps.assinaturasModulosRepositorio,
    relogio: deps.relogio,
    transacao: deps.transacao,

  const resultado = await desvincularModuloDaAssinatura({
    assinaturaId: assinaturaAtiva.value.assinaturaId,
    moduloId: paramsParse.data.moduloId,
  });

  await auditarController(request, ctx, {
    evento: "assinatura.modulo.desvinculado",
    sucesso: resultado.ok,
    entidadeTipo: "assinatura_modulo",
    entidadeId: paramsParse.data.moduloId,
    erroCodigo: !resultado.ok ? resultado.error.codigo : null,
    erroMensagem: !resultado.ok ? resultado.error.mensagem : null,
    metadados: metaAcao({
      assinaturaId: assinaturaAtiva.value.assinaturaId,
      moduloId: paramsParse.data.moduloId,
    }),
  });

  if (!resultado.ok) return sendHttpError(reply, resultado.error);

  return reply.status(200).send({
    item: {
      assinaturaId: resultado.value.assinaturaId,
      moduloId: resultado.value.moduloId,
      desabilitadoEm: resultado.value.desabilitadoEm.toISOString(),
    },
  });
}

import { z } from "zod";

function parseDateOpcional(value: unknown) {
  if (value === undefined || value === null || value === "") return undefined;
  const date = new Date(String(value));
  return date;
}

export const listarAuditoriaQuerySchema = z.object({
  evento: z.string().trim().optional(),
  eventoPrefixo: z.string().trim().optional(),
  sucesso: z
    .enum(["true", "false"])
    .optional()
    .transform((value) =>
      value === undefined ? undefined : value === "true",
    ),
  erroCodigo: z.string().trim().optional(),
  entidadeNome: z.string().trim().optional(),
  entidadeTipo: z.string().trim().optional(),
  entidadeId: z.string().trim().optional(),
  usuarioId: z.string().trim().optional(),
  origem: z.string().trim().optional(),
  requisicaoId: z.string().trim().optional(),
  agenteTipo: z.string().trim().optional(),
  criadoDe: z.preprocess(parseDateOpcional, z.date().optional()),
  criadoAte: z.preprocess(parseDateOpcional, z.date().optional()),
  textoLivre: z.string().trim().optional(),
  cursor: z.string().trim().optional(),
  limite: z.coerce.number().int().positive().max(500).optional(),
});

export type ListarAuditoriaQueryDto = z.infer<
  typeof listarAuditoriaQuerySchema
>;

export function parseListarAuditoriaQuery(input: unknown) {
  return listarAuditoriaQuerySchema.safeParse(input);
}

  return {

    evento: evento.evento,
    categoria: evento.categoria,
    sucesso: evento.sucesso,
    erroCodigo: evento.erroCodigo,
    erroMensagem: evento.erroMensagem,
    entidadeNome: evento.entidadeNome,
    entidadeTipo: evento.entidadeTipo,
    entidadeId: evento.entidadeId,
    origem: evento.origem,
    requisicaoId: evento.requisicaoId,
    enderecoIp: evento.enderecoIp,
    agenteUsuario: evento.agenteUsuario,
    agenteTipo: evento.agenteTipo,
    metadados: evento.metadados,
    criadoEm: evento.criadoEm,
  };
}

export function toListarAuditoriaResponseDto(itens: Array<EventoAuditoriaLocatario | EventoAuditoriaPlataforma>) {

import { z } from "zod";

const dateQuerySchema = z
  .string()
  .trim()
  .min(1)
  .optional()
  .transform((value) => {
    if (!value) return undefined;
    const date = new Date(value);
    return Number.isNaN(date.getTime()) ? undefined : date;
  });

export const financeiroQuerySchema = z.object({
  status: z.enum(["ABERTO", "LIQUIDADO", "TODOS"]).optional(),
  vencidas: z
    .union([z.literal("true"), z.literal("false")])
    .optional()
    .transform((value) => value === "true"),
  dataInicial: dateQuerySchema,
  dataFinal: dateQuerySchema,
  cursor: z.string().trim().min(1).optional(),
  limite: z.coerce.number().int().min(1).max(100).optional(),
});

export type FinanceiroQueryDto = z.infer<typeof financeiroQuerySchema>;

export function parseFinanceiroQuery(input: unknown) {
  return financeiroQuerySchema.safeParse(input ?? {});
}

/**
 * Cria um novo usuário no locatário autenticado.
 *
 * Permissão exigida:
 * - `tenant.usuario.criar`
 *
 * Fluxo:
 * 1. autoriza a operação;
 * 2. gera hash da senha via porta técnica oficial do runtime de auth;
 * 3. executa o caso de uso de criação no domínio;
 * 4. devolve apenas o identificador criado.
 *
 * Resposta:
 * - `201 Created` com `{ item: { usuarioId } }`
 *
 * @param request Requisição Fastify com `email`, `nome` e `senha`.
 * @param reply Resposta Fastify.
 * @returns Identificador do usuário recém-criado.
 */
export async function criarUsuarioController(request: FastifyRequest<{ Body: CriarUsuarioBodyDto }>, reply: FastifyReply) {
  const ctx = await autorizarHttp(request, reply, {
    permissao: Permissoes.TENANT_USUARIO_CRIAR,
    modulo: "USUARIOS",
  });
  if (!ctx) return;

  const bodyParse = parseCriarUsuarioBody(request.body);
  if (!bodyParse.success) {
    return responderErroValidacao(reply, bodyParse);
  }

  const body = bodyParse.data;

  const senhaHashRes = await getAuthBootstrap().runtime.emitirSessaoDeps.hashPorta.gerarHashSenha(body.senha);
  if (!senhaHashRes.ok) {
    return sendHttpError(reply, senhaHashRes.error);
  }

  const { usuariosRepositorio, transacao, relogio, idPorta } = criarDepsUsuarios();

  const criarUsuario = createFnCriarUsuario({
    usuariosRepositorio,
    transacao,
    relogio,
    idPorta,
  });

  const resultado = await criarUsuario({
    locatarioId: ctx.locatarioId,
    email: request.body.email,
    nome: request.body.nome,
    senhaHash: senhaHashRes.value,
  });

  await auditarController(request, ctx, {
    evento: "usuario.criado",
    sucesso: resultado.ok,
    entidadeTipo: "usuario",
    entidadeId: resultado.ok ? resultado.value.usuarioId : null,
    entidadeNome: request.body.nome,
    erroCodigo: !resultado.ok ? resultado.error.codigo : null,
    erroMensagem: !resultado.ok ? resultado.error.mensagem : null,
    metadados: resultado.ok
      ? metaCriacao({
          usuarioId: resultado.value.usuarioId,
          nome: request.body.nome,
          email: request.body.email,
          status: "ATIVO",
        })
      : metaAcao({
          email: request.body.email,
        }),
  });

  if (!resultado.ok) {
    return sendHttpError(reply, resultado.error);
  }

  return reply.status(201).send({
    item: {
      usuarioId: resultado.value.usuarioId,
    },
  });
}

 * - `senha` obrigatória;

 * - aqui garantimos apenas o contrato mínimo da borda.

  observacao: z.string().trim().optional().nullable(),
});

  produtoId: z.string().trim().min(1, "O campo 'produtoId' é obrigatório."),
  estoqueId: z.string().trim().optional().nullable(),
  descricao: z.string().trim().optional().nullable(),
  quantidade: z.coerce.number().positive("Quantidade deve ser maior que zero."),
  valorUnitario: z.coerce.number().nonnegative("Valor unitário não pode ser negativo.").optional(),
  valorDesconto: z.coerce.number().nonnegative().default(0),
});

  .object({
    descricao: z.string().trim().optional().nullable(),

    valorDesconto: z.coerce.number().nonnegative().optional(),
    valorAcrescimo: z.coerce.number().nonnegative().optional(),
  })
  .refine(
    (data) =>
      data.descricao !== undefined ||
      data.quantidade !== undefined ||
      data.valorUnitario !== undefined ||
      data.valorDesconto !== undefined ||
      data.valorAcrescimo !== undefined,

  );

  return {
    service: "core-kit-api",
    version: process.env.npm_package_version ?? "0.1.0",
    commit: process.env.RENDER_GIT_COMMIT ?? process.env.GIT_COMMIT ?? "local",
    env: process.env.NODE_ENV ?? "development",
  };
}

export const healthRoutes: FastifyPluginAsync = async (app) => {
  app.get("/health", async () => {
    return {
      status: "ok",
      service: "core-kit-api",
    };
  });

  app.get("/version", async () => {
    return versionInfo();
  });

  app.get("/ready", async (_request, reply) => {
    try {
      await clientePrisma.$queryRaw`SELECT 1`;

      return {
        status: "ready",
        database: "ok",
        ...versionInfo(),
      };
    } catch {
      return reply.status(503).send({
        status: "not_ready",
        database: "error",
        ...versionInfo(),
      });
    }
  });
};

 * - o escopo efetivo do locatário é sempre derivado do contexto autenticado.

 * - `GET /usuarios`
 * - `GET /usuarios/:id`
 * - `POST /usuarios`
 * - `PATCH /usuarios/:id`
 * - `POST /usuarios/:id/inativar`
 * - `POST /usuarios/:id/reativar`
 *
 * @param app Instância Fastify usada para registrar as rotas.
 * @returns Plugin assíncrono concluído.
 */
export const usuariosRoutes: FastifyPluginAsync = async (app) => {

   * Lista usuários do locatário autenticado.
   */
  app.get(
    "/usuarios",
    { preHandler: [authGuard], },
    listarUsuariosController,
  );

  /**
   * Obtém um usuário específico do locatário autenticado.
   */
  app.get<{ Params: { id: string } }>(
    "/usuarios/:id",
    { preHandler: [authGuard], },
    obterUsuarioController,
  );

  /**
   * Cria um novo usuário no locatário autenticado.
   */
  app.post<{ Body: { email: string; nome: string; senha: string } }>(
    "/usuarios",
    { preHandler: [authGuard] },
    criarUsuarioController,
  );

  /**
   * Atualiza dados cadastrais de um usuário.
   */
  app.patch<{ Params: { id: string }; Body: { email?: string; nome?: string };}>(
    "/usuarios/:id",
    { preHandler: [authGuard] },
    atualizarUsuarioController,
  );

  /**
   * Inativa um usuário existente.
   */
  app.post<{ Params: { id: string } }>(
    "/usuarios/:id/inativar",
    { preHandler: [authGuard] },
    desativarUsuarioController,
  );

  /**
   * Reativa um usuário previamente inativado.

  app.post<{ Params: { id: string } }>(
    "/usuarios/:id/reativar",
    { preHandler: [authGuard] },
    reativarUsuarioController,
  );

  /**
   *
