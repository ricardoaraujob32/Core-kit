// apps/api/__tests__/bootstrap/applicationBootstrap.test.ts

import { describe, expect, it } from "vitest";
import Fastify from "fastify";
import { runApplicationBootstrap } from "../../src/bootstrap/applicationBootstrap";

describe("applicationBootstrap", () => {
  it("deve decorar o app com auth e authorization", async () => {
    const app = Fastify();

    const bootstrap = await runApplicationBootstrap(app);

    expect(bootstrap.auth).toBeTruthy();
    expect(bootstrap.authorization).toBeTruthy();
    expect(app.applicationBootstrap.auth).toBe(bootstrap.auth);
    expect(app.applicationBootstrap.authorization).toBe(bootstrap.authorization);

    await app.close();
  });
});

// apps/api/__tests__/bootstrap/authBootstrap.test.ts

import { describe, expect, it } from "vitest";
import { createAuthBootstrap, getAuthBootstrap } from "../../src/bootstrap/authBootstrap";

describe("#1 - authBootstrap", () => {
  it("#1.1 - deve criar bootstrap de auth com runtime", () => {
    const bootstrap = createAuthBootstrap({
      eventos: {
        async registrarEvento() {},
      },
    });

    expect(bootstrap.runtime).toBeTruthy();
    expect(typeof bootstrap.runtime.emitirSessao).toBe("function");
    expect(typeof bootstrap.runtime.validarSessao).toBe("function");
    expect(typeof bootstrap.runtime.rotacionarRefresh).toBe("function");
  });

  it("#1.2 - deve expor a instância inicializada", () => {
    const bootstrap = createAuthBootstrap({
      eventos: {
        async registrarEvento() {},
      },
    });

    expect(getAuthBootstrap()).toBe(bootstrap);
  });
});
